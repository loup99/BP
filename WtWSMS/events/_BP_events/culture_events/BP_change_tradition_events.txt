namespace = bp_change_tradition_events

#Migratory Culture Tradition => Ruling Caste Tradition
bp_change_tradition_events.0001 = {
	scope = culture
	hidden = yes
	content_source = dlc_WtWSMS
	
	trigger = {
		root = { BP_has_migratory_ruling_caste_tradition_trigger = yes }
		BP_migratory_ruling_cast_tradition_too_many_counties_trigger = yes
	}
	
	immediate = {
		if = {
			limit = { has_cultural_tradition = tradition_elite_rulership }
			remove_culture_tradition = tradition_elite_rulership
			add_culture_tradition = tradition_ruling_caste
		}
		else_if = {
			limit = { has_cultural_tradition = tradition_nomadic_rulership }
			remove_culture_tradition = tradition_nomadic_rulership
			add_culture_tradition = tradition_ruling_caste
		}
		#Remove at least one MAA
		every_ruler = {
			limit = { this.culture = root }
			random_maa_regiment = { destroy_maa_regiment = yes }
		}
		#Change head determination, if not culture head is not nomadic or herder
		if = {
			limit = {
				has_cultural_pillar = head_determination_herd
				NOR = {
					this.culture_head ?= { government_has_flag = government_is_nomadic }
					this.culture_head ?= { government_has_flag = herder_government }
				}
			}
			set_culture_pillar = head_determination_domain
		}
	}
}

#Tribal Hierarchy Tradition => Hereditary Hierarchy Tradition
bp_change_tradition_events.0002 = {
	scope = culture
	hidden = yes
	content_source = dlc_WtWSMS
	
	trigger = {
		root = { has_cultural_tradition = tradition_tribal_hierarchy }
		trigger_if = { #If a culture head exists, no longer a high tribal of some sort and Classical Antiquity or later
			limit = { exists = this.culture_head }
			this.culture_head = {
				NAND = {
					has_government = tribal_government
					has_government = high_tribal_government
					has_government = nomad_government
				}
			}
			no_longer_in_tribal_era_trigger = yes
		}
		trigger_else = { has_cultural_era_or_later = culture_era_early_medieval }
	}
	
	immediate = {
		remove_culture_tradition = tradition_tribal_hierarchy
		add_culture_tradition = tradition_hereditary_hierarchy
	}
}

#Zealous Pagans Tradition Removal
bp_change_tradition_events.0003 = {
	scope = culture
	hidden = yes
	content_source = dlc_WtWSMS
	
	trigger = {
		OR = {
			root = { has_cultural_tradition = tradition_zealous_pagans }
			root = { has_cultural_tradition = tradition_zealous_pagans_non_dlc }
		}
		trigger_if = {
			limit = { exists = this.culture_head }
			NOT = { this.culture_head.faith = { has_doctrine_parameter = unreformed } }
		}
		trigger_else = {
			any_culture_county = {
				percent >= 0.50
				NOT = { this.faith = { has_doctrine_parameter = unreformed } }
			}
		}
	}
	
	immediate = {
		if = {
			limit = { has_cultural_tradition = tradition_zealous_pagans }
			remove_culture_tradition = tradition_zealous_pagans
		}
		else = { remove_culture_tradition = tradition_zealous_pagans_non_dlc }
	}
}
