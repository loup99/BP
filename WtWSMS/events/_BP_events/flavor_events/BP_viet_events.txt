# Noric smiths on the rise
bp_viet.4000 = { #WTWSMS id: VIETmisc.4000
	type = character_event
	title = bp_viet.4000.title
	desc = bp_viet.4000.desc
	
	theme = martial
	content_source = dlc_WtWSMS
	
	trigger = { is_landed = yes }

        base = 0
		modifier = {
			add = {
				every_held_title = {
					limit = {
						tier = tier_county

						NOT = { has_county_modifier = VIET_noric_smiths }
					}
					add = 0.5

						limit = { title_province = { geographical_region = custom_bavaria } }

			}
		}

		random_held_title = {
			limit = {

		}

		character = root
		animation = inspect_weapon

	option = { # Good
		name = bp_viet.4000.a

			title = VIET_noric_smiths

							modifier = VIET_noric_smiths

						add_county_modifier = {
							modifier = VIET_noric_smiths
							days = 2000
						}
					}
					20 = {
						add_county_modifier = {
							modifier = VIET_noric_smiths
							days = 3000
						}
					}
					10 = {
						add_county_modifier = {
							modifier = VIET_noric_smiths
							days = 4000
						}
					}
				}
			}
		}
	}
	option = { # Why didn't you invest in eastern Noricum?
		name = bp_viet.4000.b
		flavor = VIET_invest_in_eastern_noricum

			stewardship < 10

# Legends of Noreia
bp_viet.4001 = { #WTWSMS id: VIETmisc.4001
	type = character_event
	title = bp_viet.4001.title
	desc = bp_viet.4001.desc
	
	theme = realm
	content_source = dlc_WtWSMS
	
	trigger = { is_landed = yes }
	
	weight_multiplier = {
        base = 0
		modifier = {
			add = {
				every_held_title = {
					limit = {
						tier = tier_county
						is_landless_type_title = no
						culture = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_noric } }
						NOT = { has_county_modifier = VIET_noric_legendary_capital }
					}
					add = 0.2
					# Makes much more sense in historical Noricum
					if = {
						limit = { title_province = { geographical_region = custom_bavaria } }
						add = 0.8
					}
				}
				#Don't make it happen too much
				if = {
					limit = { has_global_variable = VIET_noreia_found }
					multiply = 0.5
				}
				#Make it less likely for AI
				if = {
					limit = { is_ai = yes }
					multiply = 0.5
				}
			}
		}
	}
	
	immediate = {
		set_global_variable = VIET_noreia_found
		add_prestige = medium_prestige_gain
		random_held_title = {
			limit = {
				tier = tier_county
				is_landless_type_title = no
				culture = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_noric } }
				NOT = { has_county_modifier = VIET_noric_legendary_capital }
			}
			save_scope_as = noric_county
			
			add_county_modifier = {
				modifier = VIET_noric_legendary_capital
				years = 50
			}
		}
		random_courtier_or_guest = {
			weight = {
				base = 0
				modifier = { add = diplomacy }
				modifier = { add = { value = learning multiply = 2 } }
			}
			limit = { is_available_ai_adult = yes }
			save_scope_as = scholar
		}
	}
	
	left_portrait = {
		character = scope:scholar
		animation = reading
	}
	right_portrait = {
		character = root
		animation = personality_forgiving
	}
	option = { # Good
		name = bp_viet.4001.a
		stress_impact = {
			arrogant = minor_stress_impact_loss
			trusting = miniscule_stress_impact_loss
		}
	}
	option = { # Disbelief
		name = bp_viet.4001.b
		stress_impact = {
			paranoid = minor_stress_impact_loss
			cynical = minor_stress_impact_loss
			arrogant = minor_stress_impact_gain
		}
	}
	option = { # Scholarly disbelief
		name = bp_viet.4001.c
		trigger = {
			learning >= 13
		}
		show_as_unavailable = {
			learning < 13
		}
		add_prestige = miniscule_prestige_gain
		stress_impact = {
			honest = minor_stress_impact_loss
			cynical = minor_stress_impact_loss
			arrogant = minor_stress_impact_gain
		}
	}
}

scripted_effect bp_viet.4002.noric_herbalist_gifter = {
	# Find a noble Noric guest
	random_courtier_or_guest = {
		limit = {
			culture = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_noric } }
			has_dynasty = yes
			is_courtier = no
		}
		save_scope_as = gifter
	}
	# If not, search for a wandering character
	if = {
		limit = { NOT = { exists = scope:gifter } }
		find_pool_characters_effect = yes
		random_in_list = {
			list = potential_pool_characters
			limit = {
				culture = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_noric } }
				has_dynasty = yes
			}
			save_scope_as = gifter
		}
		add_visiting_courtier = scope:gifter
	}
	# If still not, make one
	if = {
		limit = { NOT = { exists = scope:gifter } }
		
		root.culture = { save_scope_as = faith }
		root.faith = { save_scope_as = faith }
		
		#Use realm if possible
		random_realm_county = {
			limit = { culture = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_noric } } }
			culture = { save_scope_as = culture }
			faith = { save_scope_as = faith }
		}
		create_character = {
			location = root.capital_province
			faith = scope:faith
			culture = scope:culture
			dynasty = generate
			template = pool_repopulate_learning
			save_scope_as = gifter
		}
		add_visiting_courtier = scope:gifter
	}
}

# Handle divergent cultures getting treated like Bosporans
scripted_trigger bp_viet.4004.bosporan_culture_trigger = {
	# Greek group
	has_cultural_pillar = heritage_byzantine
	OR = {
		this = culture:bosporan
		# Bosporan or steppe-descended
		any_parent_culture_or_above = {
			OR = {
				this = culture:bosporan
				has_cultural_parameter = heritage_group_eurasian_steppe
			}
		}
	}
}

scripted_effect bp_viet.4004.greek_traveler = {
	# Find a Greek guest
	random_courtier_or_guest = {
		limit = {
			culture = {
				has_cultural_pillar = heritage_byzantine
				bp_viet.4004.bosporan_culture_trigger = no
			}
			is_courtier = no
		}
		save_scope_as = traveler
	}
	# If not, search for a wandering character
	if = {
		limit = { NOT = { exists = scope:traveler } }
		find_pool_characters_effect = yes
		random_in_list = {
			list = potential_pool_characters
			limit = {
				culture = {
					has_cultural_pillar = heritage_byzantine
					bp_viet.4004.bosporan_culture_trigger = no
				}
			}
			save_scope_as = traveler
			add_visiting_courtier = scope:traveler
		}
	}
	# If still not, make one
	if = {
		limit = { NOT = { exists = scope:traveler } }
		
		create_character = {
			location = root.capital_province
			culture = culture:greek
			template = traveler_character
			save_scope_as = traveler
		}
		add_visiting_courtier = scope:traveler
	}
}
