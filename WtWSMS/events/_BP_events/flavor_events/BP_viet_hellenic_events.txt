scripted_trigger hellenic_or_syncretic_trigger = {
	faith = {
		OR = {
			has_doctrine = tenet_hellenic_syncretism_pagan_only
			has_doctrine = gnostic_syncretism_hellenism
			religion = religion:hellenism_religion
		}
	}
}

# Trigger for classical myth events
scripted_trigger hellenic_or_poetic_trigger = {
	OR = {
		religion = religion:hellenism_religion
		
		AND = {
			# Knows the classics
			has_trait = lifestyle_poet
			# Is Greek/Roman/Syncretic
			OR = {
				faith = {
					OR = {
						has_doctrine = tenet_hellenic_syncretism_pagan_only
						has_doctrine = gnostic_syncretism_hellenism
					}
				}
				OR = {
					knows_language = language_latin
					knows_language = language_greek
				}
			}
		}
	}
}

scripted_trigger hellenic_502_bastard_child = {
	age < 1
	NOR = {
		has_character_flag = VIET_hellenic_502_involved
		has_trait = twin # Easier this way
	}
	exists = mother
	exists = father
	hellenic_or_syncretic_trigger = yes
	father = {
		has_dynasty = yes
		is_married = yes
	}
	mother = {
		NOT = { has_character_flag = VIET_hellenic_502_involved }
		is_attracted_to_men = yes
	}
	OR = {
		# Actually a bastard
		this.father != this.real_father
		# Father suspects unfaithfulness
		mother = { has_trait = adulterer }
		mother = { any_secret = { secret_type = secret_lover } }
	}
}
