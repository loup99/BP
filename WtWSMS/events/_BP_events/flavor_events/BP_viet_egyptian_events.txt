namespace = egyptian

scripted_trigger egyptian_or_syncretic_trigger = {
	faith = {
		OR = {
			religion = religion:kushitism_religion
			has_doctrine = tenet_egyptian_syncretism_pagan_only
		}
	}
}
scripted_trigger egyptian_culture_or_syncretic_trigger = {
	OR = {
		culture = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:ancient_egyptian } }
		culture = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:old_nubian } }
		egyptian_or_syncretic_trigger = yes
	}
}

scripted_trigger has_named_egyptian_sin = {
	OR = {
		AND = { has_trait = lustful faith = { trait_is_sin = lustful } }
		AND = { has_trait = gluttonous faith = { trait_is_sin = gluttonous } }
		AND = { has_trait = greedy faith = { trait_is_sin = greedy } }
		AND = { has_trait = lazy faith = { trait_is_sin = lazy } }
		AND = { has_trait = wrathful faith = { trait_is_sin = wrathful } }
		AND = { has_trait = arrogant faith = { trait_is_sin = arrogant } }
		AND = { has_trait = deceitful faith = { trait_is_sin = deceitful } }
		AND = { has_trait = craven faith = { trait_is_sin = craven } }
		AND = { has_trait = ambitious faith = { trait_is_sin = ambitious } }
		AND = { has_trait = arbitrary faith = { trait_is_sin = arbitrary } }
		AND = { has_trait = lustful faith = { trait_is_sin = cynical } }
		AND = { has_trait = paranoid faith = { trait_is_sin = paranoid } }
		AND = { has_trait = callous faith = { trait_is_sin = callous } }
		AND = { has_trait = sadistic faith = { trait_is_sin = sadistic } }
		AND = { has_trait = vengeful faith = { trait_is_sin = vengeful } }
	}
}

scripted_effect egyptian_156_expel_evil_effect = {
	every_character_trait = {
		limit = { root.faith = { trait_is_sin = prev } }
		add_to_list = traits_to_remove
	}
	
	every_in_list = {
		list = traits_to_remove
		save_temporary_scope_as = temp_trait
		random_list = {
			# Replace the trait with a virtue
			33 = {
				root = { remove_trait = scope:temp_trait }
				random_trait = {
					limit = {
						has_trait_category = personality
						root.faith = { trait_is_virtue = prev }
						NOT = { root = { has_trait = prev } }
					}
					save_temporary_scope_as = temp_trait
				}
				root = { add_trait = scope:temp_trait }
			}
			# Failure, add stress
			66 = {
				root = {
					send_interface_toast = {
						type = event_stress_bad
						title = egyptian.156.failure
						add_stress = major_stress_gain
						left_icon = root
					}
				}
			}
		}
	}
}
