scripted_effect remove_a_tolbiac_modifier_effect = {
	if = {
		limit = { has_character_modifier = battle_of_tolbiac_$NUM$ }
		remove_character_modifier = battle_of_tolbiac_$NUM$
	}
}

	if = {

#Choice Effects
scripted_effect battle_of_tolbiac_picked_wrong_effect = {
	add_piety_level = -1
	add_prestige_level = -1
	
	#Possible random bad effect
	random_list = {
		100 = {} #Nothing
		#Well, I was just being crazy
		100 = { add_stress = -25 }
		50 = { add_stress = -50 }
		25 = { add_stress = -100 }
		#This was possibly Divine Providence, are you sure?
		100 = { add_stress = 50 }
		50 = { add_stress = 100 }
		25 = { add_stress = 150 }
		10 = { add_stress = 200 }
		#Your personality undergoes a change from doing this
		25 = { trigger = { has_trait = fickle } remove_trait = fickle add_trait = stubborn }
		25 = { trigger = { has_trait = brave } remove_trait = brave add_trait = craven }
		25 = { trigger = { has_trait = trusting } remove_trait = trusting add_trait = paranoid }
		25 = { trigger = { NOT = { has_trait = lunatic_1 } } add_trait = lunatic_1 }
		#And there's a tiny chance when you welch God strikes you down/the stress _really_ gets to you...
		5 = { trigger = { NOT = { has_trait = faltering_heart } } add_trait = faltering_heart }
		1 = { death = natural }
	}
}
scripted_effect battle_of_tolbiac_picked_right_new_god_effect = {
	$OLD_FAITH$ = { save_scope_as = old_faith }
	this = { save_scope_as = actor }
	
	set_character_faith = root.var:the_tolbiac_faith
	
	if = { limit = { has_trait = cynical } remove_trait = cynical }
	add_trait = zealous
	add_trait = faith_warrior
	
	if = {
		limit = {
			number_of_commander_traits < 3
			NOT = { has_trait = holy_warrior }
		}
		add_trait = holy_warrior
		add_trait_xp = {
			trait = holy_warrior
			value = { 25 75 }
		}
	}
	
	#Duplicate, make more powerful, and fire without rescoping
	hidden_effect = {
		trigger_event = { id = faith_conversion.0001 } # Convert Capital County
		trigger_event = { id = faith_conversion.0003 } # Convert close family
		trigger_event = { id = faith_conversion.1002 } # Player-only narrative-notification event
		trigger_event = { id = faith_conversion.1101 } # Notify vassals of conversion
	}
	
	custom_tooltip = {
		text = 50_percent_chance_every_realm_county_flips
		every_held_title = {
			limit = {
				is_landless_type_title = no
				tier = tier_county
				NOT = { this.faith = root.var:the_tolbiac_faith }
			}
			random = {
				chance = 50
				set_county_faith = root.var:the_tolbiac_faith
			}
		}
	}
	custom_tooltip = {
		text = 50_percent_chance_every_nonzealous_vassal_flips
		every_vassal = {
			limit = {
				NOT = { this.faith = root.var:the_tolbiac_faith }
				NOT = { has_trait = zealous }
			}
			random = {
				chance = 50
				set_character_faith = root.var:the_tolbiac_faith
			}
		}
	}
	
	$NEW_FAITH$ = { change_fervor = {value = 10 desc = battle_of_tolbiac_won } }
	$OLD_FAITH$ = { change_fervor = {value = -5 desc = battle_of_tolbiac_lost } }
	
	set_global_variable = battle_of_tolbiac_happened
}
scripted_effect battle_of_tolbiac_picked_right_old_gods_effect = {
	add_piety_level = 1
	add_prestige_level = 1
	
	add_trait = faith_warrior
	
	if = {
		limit = {
			number_of_commander_traits < 3
			NOT = { has_trait = holy_warrior }
		}
		add_trait = holy_warrior
		add_trait_xp = {
			trait = holy_warrior
			value = { 25 75 }
		}
	}
	
	$NEW_FAITH$ = { change_fervor = { value = -5 desc = battle_of_tolbiac_lost } }
	$OLD_FAITH$ = { change_fervor = { value = 10 desc = battle_of_tolbiac_won } }
}

#You get a chance to have the Battle of Tolbiac Event
battle_of_tolbiac.0001 = {
	type = character_event
	theme = faith
	content_source = dlc_WtWSMS
	override_background = { reference = battlefield }
	
	
	title = battle_of_tolbiac.0001.t
	
	desc = {
		desc = battle_of_tolbiac.0001.desc_opening
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:spouse_faith }
				desc = battle_of_tolbiac.0001.desc_spouse
			}
			triggered_desc = {
				trigger = { exists = scope:concubine_faith }
				desc = battle_of_tolbiac.0001.desc_concubine
			}
			triggered_desc = {
				trigger = { exists = scope:consort_faith }
				desc = battle_of_tolbiac.0001.desc_consort
			}
			triggered_desc = {
				trigger = { exists = scope:capital_faith }
				desc = battle_of_tolbiac.0001.desc_capital
			}
			desc = battle_of_tolbiac.0001.desc_county
		}
		desc = battle_of_tolbiac.0001.desc_closing
	}
	
	immediate = {
		add_character_flag = battle_of_tolbiac_happened_personal #Only ever once per character
		save_scope_as = the_char
		
		if = {
			limit = { any_spouse = { this.faith = root.var:the_tolbiac_faith } }
			ordered_spouse = {
				order_by = "reverse_opinion(scope:the_char)"
				save_scope_as = spouse_faith
			}
		}
		else_if = {
			limit = { any_concubine = { this.faith = root.var:the_tolbiac_faith } }
			ordered_concubine = {
				order_by = "reverse_opinion(scope:the_char)"
				save_scope_as = concubine_faith
			}
		}
		else_if = {
			limit = { any_consort = { this.faith = root.var:the_tolbiac_faith } }
			ordered_consort = {
				order_by = "reverse_opinion(scope:the_char)"
				save_scope_as = consort_faith
			}
		}
		else_if = {
			limit = { this.capital_county.faith = root.var:the_tolbiac_faith }
			save_scope_as = capital_faith
		}
	}
	
	left_portrait = {
		character = ROOT
		animation = personality_cynical
	}
	
	option = {
		name = battle_of_tolbiac.0001.accept_the_new_god
		
		#Special Flavor depending on the religion
		flavor = {
			first_valid = {
				#Clovis Plea to the heavens
				triggered_desc = {
					trigger = {
						root.culture = culture:old_frankish
						root.var:the_tolbiac_faith.religion = religion:christianity_religion
					}
					desc = battle_of_tolbiac.0001.accept_the_new_god.flavor.clovis
				}
				#If already cynical
				triggered_desc = {
					trigger = { has_trait = cynical }
					desc = battle_of_tolbiac.0001.accept_the_new_god.flavor.cynical
				}
				#The Our Father
				triggered_desc = {
					trigger = { root.var:the_tolbiac_faith.religion = religion:christianity_religion }
					desc = battle_of_tolbiac.0001.accept_the_new_god.flavor.christian
				}
				desc = battle_of_tolbiac.0001.accept_the_new_god.generic
			}
		}
		
		#You can't be zealous now
		if = { limit = { has_trait = zealous } remove_trait = zealous add_trait = cynical }
		if = { limit = { has_trait = holy_warrior } remove_trait = holy_warrior }
		
		custom_tooltip = {
			text = get_random_battle_modifier
			
			#Add Random Character Modifier via random list
			add_tolbiac_modifier_effect = yes
		}
		
		#Flags for bookkeeping
		add_character_flag = accepted_the_new_god
		add_character_flag = battle_of_tolbiac_in_progress_char
		
		ai_chance = { base = 50 }
	}
	option = {
		name = battle_of_tolbiac.0001.keep_the_old_gods
		if = {
			limit = { NOT = { has_trait = cautious_leader } }
			add_trait = cautious_leader
			add_trait_xp = { trait = cautious_leader value = { 25 50 } }
		}
		else = { add_trait_xp = { trait = cautious_leader value = 25 } }
		
		flavor = {
			first_valid = {
				triggered_desc = {
					trigger = { has_trait = cynical }
					desc = battle_of_tolbiac.0001.keep_the_old_gods.flavor.cynical
				}
				desc = battle_of_tolbiac.0001.keep_the_old_gods.generic
			}
		}
		
		#Flags for bookkeeping
		add_character_flag = kept_the_old_gods
		add_character_flag = battle_of_tolbiac_in_progress_char
		
		ai_chance = { base = 50 }
	}
	option = { #Ends the chain right here.
		name = battle_of_tolbiac.0001.rely_on_myself
		
		flavor = battle_of_tolbiac.0001.rely_on_myself.generic
		
		add_prestige = 150
		add_piety = -50
		
		ai_chance = { base = 0 }
	}
}

battle_of_tolbiac.0009 = {
	type = character_event
	theme = faith
	content_source = dlc_WtWSMS
	override_background = { reference = throne_room_scope }
	
	
	title = battle_of_tolbiac.0009.t
	
	desc = battle_of_tolbiac.0009.desc
	
	immediate = {
		scope:battle_of_tolbiac_winner = { save_scope_as = background_throne_room_scope }
	}
	
	right_portrait = {
		character = scope:battle_of_tolbiac_winner
		animation = prayer
	}
	
	option = {
		trigger = {
			scope:battle_of_tolbiac_winner.faith = {
				faith_hostility_level = { target = root.faith value <= faith_astray_level }
			}
		}
		name = battle_of_tolbiac.0009.good
	}
	option = {
		fallback = yes
		name = battle_of_tolbiac.0009.ok
	}
	option = {
		trigger = {
			scope:battle_of_tolbiac_winner.faith = {
				faith_hostility_level = { target = root.faith value > faith_astray_level }
			}
			NOT = { root.faith = scope:old_faith }
		}
		name = battle_of_tolbiac.0009.bad
	}
	option = {
		trigger = { root.faith = scope:old_faith }
		name = battle_of_tolbiac.0009.terrible
	}
}
