scripted_effect remove_character_flag_if_has_it_effect = {
	if = {
		limit = { has_character_flag = $FLAG$ }
		remove_character_flag = $FLAG$
	}
}
scripted_effect jumpstart_culture_progress_effect = {
	$SOURCE_CULTURE$ = {
		every_known_innovation = {
			limit = {
				innovation_era = culture_era_tribal
				has_innovation_flag = tribal_era_regular
			}
			save_temporary_scope_as = the_innovation
			if = {
				limit = { NOT = { $TARGET_CULTURE$ = { has_innovation = scope:the_innovation } } }
				add_to_list = innovations_to_add
			}
		}
	}
	every_in_list = {
		list = innovations_to_add
		save_temporary_scope_as = the_innovation
		$TARGET_CULTURE$ = { add_innovation_progress = { value = 5 target = scope:the_innovation } }
	}
	$TARGET_CULTURE$ = {
		if = {
			limit = { NOT = { has_cultural_era_or_later = culture_era_classical_antiquity } }
			join_era = culture_era_classical_antiquity
		}
	}
}
scripted_effect give_cash_effect = {
	if = {
		limit = { has_treasury = no }
		add_gold = migration_gold_gift_without_treasury
	}
	else = {
		add_gold = migration_gold_gift_with_treasury
		add_treasury = migration_treasury
	}
}

scripted_effect apply_traditions_to_ruler_culture_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { culture = { has_cultural_pillar = head_determination_herd } } }
			culture = { set_culture_pillar = head_determination_herd }
		}
	}
	#Tradition padding
	if = {
		limit = { culture = { has_cultural_tradition = tradition_ruling_caste } }
		culture = { remove_culture_tradition = tradition_ruling_caste }
	}
	if = {
		limit = { culture = { has_cultural_tradition = tradition_cranial_deformation } }
		culture = {
			remove_culture_tradition = tradition_cranial_deformation
			add_culture_tradition = tradition_nomadic_rulership_cranial_deformation
		}
	}
	else_if = {
		limit = { has_character_flag = nomadic_invader }
		culture = { add_culture_tradition = tradition_nomadic_rulership }
	}
	else = { culture = { add_culture_tradition = tradition_elite_rulership } }
}

	trigger = {
		has_dlc_feature = diverge_culture #Safety
		BP_diverge_culture_during_migration_trigger = yes
		#Not already an elite culture tradition
		NOR = {
			culture = { has_cultural_tradition = tradition_nomadic_rulership }
			culture = { has_cultural_tradition = tradition_nomadic_rulership_cranial_deformation }
			culture = { has_cultural_tradition = tradition_elite_rulership }
		}
	}
	immediate = {
		#Pick who actually diverges the culture
		if = { #I am the culture head; the peasants diverge!
			limit = {
				OR = {
					root = culture.culture_head
					has_character_flag = was_culture_head
				}
			}
			ordered_ruler = {
				limit = {
					NOT = { this = root }
					NOT = { this = { any_liege_or_above = { this = root } } }
					this.culture = root.culture
					is_landed = yes
				}
				order_by = primary_title.tier
				this = { save_scope_as = culture_former }
			}
		}
		else = { #I am not the culture head; I change and the peasants shift
			root = { save_scope_as = culture_former }
		}
		#Fallback; nobody was picked. Shouldn't happen, but might with nomad wonkiness
		if = {
			limit = { NOT = { exists = scope:culture_former } }
			root = { save_scope_as = culture_former }
		}
		scope:culture_former = { save_scope_as = background_throne_room_scope }
	}
	after = {
		hidden_effect = {
			remove_character_flag_if_has_it_effect = { FLAG = was_culture_head }
		}

	}
