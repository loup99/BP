#Fall of Rome
bp_narrative_events.0001 = { #WTWSMS id: zusk_event.30
	title = bp_narrative_events.0001.name
	desc = bp_narrative_events.0001.desc
	type = character_event
	window = fullscreen_event
	theme = battle
	content_source = dlc_WtWSMS
	override_background = { reference = sack_of_city }

	immediate = {
		title:c_roma.holder.top_liege = { save_scope_as = barbarian_ruler }
		scope:barbarian_ruler = { trigger_event = { id = bp_narrative_events.0003 days = 2 delayed = yes } }
	}

	right_portrait = {
		character = scope:barbarian_ruler
		animation = war_over_win
	}

	option = { #General Response
		fallback = yes
		name = bp_narrative_events.0001.general_response
		trigger = {
			top_liege = {
				NOR = {
					has_title = title:e_byzantium
					has_title = title:h_eastern_roman_empire
				}
			}
			NOR = { religion = religion:germanic_religion }
			NOR = { religion = religion:hellenism_religion }
			NOT = { this = scope:barbarian_ruler }
			NOT = { this = character:93 }
			NOT = { this = character:175110 }
			NAND = {
				religion = religion:christianity_religion
				this.faith = { considers_astray_trigger = { TARGET_FAITH = faith:nicene } }
			}
		}
	}
	option = { #Germanic Pagans
		name = bp_narrative_events.0001.germanic_religion_response
		trigger = {
			top_liege = {
				NOR = {
					has_title = title:e_byzantium
					has_title = title:h_eastern_roman_empire
				}
			}
			NOT = { this = scope:barbarian_ruler }
			NOT = { this = character:93 }
			NOT = { this = character:175110 }
			religion = religion:germanic_religion
		}
		add_piety = 100
	}
	option = { #Christians harmonized with the Imperial church
		name = bp_narrative_events.0001.imperial_christian_response
		trigger = {
			top_liege = {
				NOR = {
					has_title = title:e_byzantium
					has_title = title:h_eastern_roman_empire
				}
			}
			NOT = { this = scope:barbarian_ruler }
			NOT = { this = character:93 }
			NOT = { this = character:175110 }
			AND = {
				religion = religion:christianity_religion
				this.faith = { considers_astray_trigger = { TARGET_FAITH = faith:nicene } }
			}
		}
		add_piety = -100
	}
	option = { #Hellenists
		name = bp_narrative_events.0001.imperial_christian_response
		trigger = {
			top_liege = {
				NOR = {
					has_title = title:e_byzantium
					has_title = title:h_eastern_roman_empire
				}
			}
			NOT = { this = scope:barbarian_ruler }
			NOT = { this = character:93 }
			NOT = { this = character:175110 }
			religion = religion:hellenism_religion
		}
		add_piety = -100
	}
	option = { #Odoacer
		name = bp_narrative_events.0001.odoacer_response
		trigger = { this = scope:barbarian_ruler }
		add_piety = 100
		add_prestige = 100
	}
	option = { #Genseric
		name = bp_narrative_events.0001.genseric_response
		trigger = { this = character:93 }
	}
	option = { #Julius Nepos
		name = bp_narrative_events.0001.nepos_response
		trigger = { this = character:175110 }
		if = {
			limit = { NOT = { exists = title:h_western_roman_empire.holder } }
			root = { save_scope_as = the_last_wre_holder }
			trigger_event = { id = bp_narrative_events.0002 }
		}
	}
	option = { #Byzantine Option
		name = bp_narrative_events.0001.byzantine_response
		trigger = {
			top_liege = {
				OR = {
					has_title = title:e_byzantium
					has_title = title:h_eastern_roman_empire
				}
			}
			NOT = { this = scope:barbarian_ruler }
			NOT = { this = character:93 }
			NOT = { this = character:175110 }
		}
		add_piety = -100
		add_prestige = -100
	}
}

#Nepos becomes Western Emperor
bp_narrative_events.0002 = { #WTWSMS id: zusk_event.26
	title = bp_narrative_events.0002.name
	desc = bp_narrative_events.0002.desc
	type = character_event
	theme = realm
	content_source = dlc_WtWSMS
	
	right_portrait = {
		character = root
		animation = personality_honorable
	}
	artifact = {
		target = scope:diadem_artifact
		position = lower_center_portrait
	}
	
	immediate = {
		# Find the Regalia
		random_artifact = {
			limit = { has_variable = western_roman_imperial_diadem }
			save_scope_as = diadem_artifact
			if = {
				limit = { artifact_owner != root }
				artifact_owner = { save_scope_as = regalia_owner }
			}
		}
	}
	artifact = {
		target = scope:diadem_artifact
		position = lower_center_portrait
	}
	
	option = {
		trigger = { root = scope:the_last_wre_holder }
		name = bp_narrative_events.0002.a
		get_title = title:h_western_roman_empire
		convert_to_roman_government_effect = yes
		every_vassal_or_below = {
			limit = {
				OR = {
					government_has_flag = government_is_subroman_administrative
					government_has_flag = government_is_subroman_dynastic
				}
			}
			convert_to_roman_government_effect = yes
		}
		scope:diadem_artifact = { set_owner = { target = root } }
		if = {
			limit = { root ?= scope:the_last_wre_holder }
			every_player = {
				limit = {
					NOT = { root = scope:the_last_wre_holder }
					is_ai = no
				}
				trigger_event = { id = bp_narrative_events.0002 }
			}
		}
	}
	option = {
		trigger = { NOT = { root = scope:the_last_wre_holder } }
		name = bp_narrative_events.0002.a
		show_as_tooltip = { scope:the_last_wre_holder = { get_title = title:h_western_roman_empire } }
	}
	option = {
		trigger = { NOT = { root = scope:the_last_wre_holder } }
		name = bp_narrative_events.0002.b
		fallback = yes
		show_as_tooltip = { scope:the_last_wre_holder = { get_title = title:h_western_roman_empire } }
	}
}

# Handling the Western Imperial Regalia after the Fall of Rome
bp_narrative_events.0003 = { #WTWSMS id: imperial_legacy.42
	title = bp_narrative_events.0003.name
	desc = bp_narrative_events.0003.desc
	type = character_event
	
	theme = coronation
	content_source = dlc_WtWSMS
	
	trigger = {
		any_artifact = {
			has_variable = western_roman_regalia
			artifact_owner = {
				OR = {
					this = root
					top_liege = root
					is_imprisoned_by = root
				}
			}
		}
	}
	
	immediate = {
		# Find the Regalia
		random_artifact = {
			limit = {
				has_variable = western_roman_regalia
				artifact_owner = {
					OR = {
						this = root
						top_liege = root
						is_imprisoned_by = root
					}
				}
			}
			save_scope_as = regalia_artifact
			artifact_owner = { save_scope_as = regalia_owner }
		}
		
		title:e_wre.holder ?= { save_scope_as = wre_holder }
		title:h_western_roman_empire.holder ?= { save_scope_as = wre_holder }
		
		title:e_byzantium.holder ?= { save_scope_as = ere_holder }
		title:h_eastern_roman_empire.holder ?= { save_scope_as = ere_holder }
		
		hidden_effect = {
			# Add claims
			scope:wre_holder ?= { add_personal_artifact_claim = scope:regalia_artifact }
			scope:ere_holder ?= { add_personal_artifact_claim = scope:regalia_artifact }
		}
	}
	
	left_portrait = {
		character = scope:regalia_owner
		animation = obsequious_bow
		camera = camera_event_very_left_further_back
	}
	artifact = {
		target = scope:regalia_artifact
		position = lower_center_portrait
	}
	right_portrait = {
		character = root
		animation = personality_bold
	}
	lower_left_portrait = scope:wre_holder
	lower_right_portrait = scope:ere_holder
	
	# Keep it
	option = {
		name = bp_narrative_events.0003.a
		add_legitimacy = medium_legitimacy_gain
		add_prestige = major_prestige_gain
		scope:regalia_artifact = {
			set_owner = {
				target = root
				history = {
					type = conquest
					actor = scope:regalia_owner
					recipient = root
				}
			}
		}
		
		# Odoacer didn't do this
		ai_chance = {
			base = 100 
			modifier = {
				factor = 0
				root = character:55
			}
		}
	}
	
	# Send to ERE
	option = {
		name = bp_narrative_events.0003.b
		trigger = { exists = scope:ere_holder }
		
		add_trait = roman_nostalgia
		if = {
			limit = { has_legitimacy = yes }
			add_legitimacy = major_legitimacy_gain
		}
		
		scope:regalia_artifact = {
			add_artifact_history = {
				type = conquest
				actor = scope:regalia_owner
				recipient = root
			}
			set_owner = {
				target = scope:ere_holder
				history = {
					type = given
					actor = root
					recipient = scope:ere_holder
				}
			}
		}
		reverse_add_opinion = {
			target = scope:ere_holder
			modifier = gift_artifact_opinion
			opinion = 100
		}
		
		# This is what Odoacer did historically
		ai_chance = {
			base = 300
		}
	}
	
	# Send to WRE
	option = {
		name = bp_narrative_events.0003.c
		trigger = { exists = scope:wre_holder }
		
		add_trait = roman_nostalgia
		add_legitimacy = medium_legitimacy_gain
		
		scope:regalia_artifact = {
			add_artifact_history = {
				type = conquest
				actor = scope:regalia_owner
				recipient = root
			}
			set_owner = {
				target = scope:wre_holder
				history = {
					type = given
					actor = root
					recipient = scope:wre_holder
				}
			}
		}
		reverse_add_opinion = {
			target = scope:wre_holder
			modifier = gift_artifact_opinion
			opinion = 100
		}
		
		ai_chance = {
			base = 200
			modifier = {
				factor = 0
				root = character:55
			}
		}
	}
}
