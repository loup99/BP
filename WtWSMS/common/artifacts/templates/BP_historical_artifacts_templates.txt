zoroastrian_relic_template = {
	can_equip = { always = yes }
	
	can_benefit = { has_religion = religion:zoroastrianism_religion }
	
	fallback = { court_grandeur_baseline_add = 3 }
	
	unique = yes
}

gurzil_template = {
	can_equip = { always = yes }
	
	can_benefit = {
		OR = {
			has_religion = religion:north_african_religion
			has_religion = religion:kushitism_religion
			faith = { has_doctrine = tenet_egyptian_syncretism_pagan_only }
		}
	}
	
	fallback = { monthly_martial_lifestyle_xp_gain_mult = 0.05 }
	
	unique = yes
}

roman_crown_template = {
	can_equip = {
		trigger_if = {
			limit = {
				has_ach_dlc_trigger = yes
				coronation_proper_artifact_crown_trigger = yes
				NOT = { scope:artifact = { category = court } }
			}
			NOT = { has_realm_law = uncrowned }
		}
	}

	# can this character benefit from the full modifiers of the artifact?
	can_benefit = {
		OR = {
			culture = { has_cultural_parameter = heritage_group_romance }
			culture = { has_cultural_pillar = heritage_byzantine }
			is_roman_emperor_trigger = yes
			any_liege_or_above = {
				is_roman_emperor_trigger = yes
			}
		}
	}

	# if a given character does not pass the "can_benefit" trigger then this modifier will be applied instead.
	fallback = {
		court_grandeur_baseline_add = 4
		monthly_dynasty_prestige_mult = 0.05
	}

	unique = yes
}
