				trigger = { rite_has_parameter = liturgical_language_latin }
				desc = tenet_pursuit_of_power_latin_name

			triggered_desc = {
				trigger = { rite_has_parameter = liturgical_language_greek }
				desc = tenet_pursuit_of_power_greek_name

			desc = tenet_pursuit_of_power_name
		}
	}
	desc = tenet_pursuit_of_power_desc
	##### Changed from Vanilla ^^^^^

	piety_cost = {

			limit = {
				OR = {
					rite_has_doctrine = pagan_hostility_doctrine

				}
			}
			value = faith_tenet_cost_mid
		}
		else_if = {
			limit = {
				rite_religion_tag = judaism_religion

		else = {
			value = faith_tenet_cost_massive
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_pursuit_of_power }

		}
	}

	is_shown = {
		OR = {
			has_dlc_feature = all_under_heaven
			has_dlc_feature = roads_to_power
		}
	}

	can_pick_as_personal_tenet = {
		NOT = { has_trait = content }
