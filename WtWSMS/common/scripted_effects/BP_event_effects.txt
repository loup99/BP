### Formation of the Aures Kingdom
form_k_aures_effect = {
	title:k_auresia = {
		set_coa = title:d_auresia
		copy_title_history = title:d_auresia
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:k_auresia = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	title:k_auresia.holder = { destroy_title = title:d_auresia }
	#Make the kingdom of Auresia part of some de jure empire
	hidden_effect = {
		title:k_auresia = { set_de_jure_liege_title = title:e_north_africa }
	}
}

### Zoroastrian Events
legalize_caucasian_christianity_global_effect = {
	rite:armenian_rite = { add_rite_doctrine = special_doctrine_legalized_faith_persia }
	rite:georgian_rite = { add_rite_doctrine = special_doctrine_legalized_faith_persia }
	rite:albanian_apostolic = { add_rite_doctrine = special_doctrine_legalized_faith_persia }
}
caucasian_christian_legalization_local_effect = {
	legalize_caucasian_christianity_global_effect = yes
	add_piety_level = -1
	#The clerics are angry
	every_vassal_or_below = {
		limit = {
			religion = religion:zoroastrianism_religion
			rite = { rite_has_doctrine = doctrine_major_branch_behdin }
		}
		if = {
			limit = { government_has_flag = government_is_theocracy }
			add_opinion = {
				modifier = angry_opinion
				target = root
				opinion = -20
			}
		}
	}
	custom_description = {
		text = every_court_chaplain_is_angry
		every_vassal_or_below = {
			limit = {
				religion = religion:zoroastrianism_religion
				rite = { rite_has_doctrine = doctrine_major_branch_behdin }
			}
			every_councillor = {
				limit = { has_council_position = councillor_court_chaplain }
				add_opinion = {
					modifier = angry_opinion
					target = root
					opinion = -20
				}
			}
		}
	}
	#All the Christians in the Empire are relieved
	custom_description = {
		text = every_christian_in_the_country_is_relieved

			limit = {
				OR = {

				modifier = legalized_my_faith

			if = {
				limit = {
					OR = {
						rite = rite:armenian_rite
						rite = rite:georgian_rite
						rite = rite:albanian_apostolic
					}
				}
				add_opinion = {
					modifier = legalized_my_faith
					years = 50
					target = root
				}
			}
			every_courtier_or_guest = {
				limit = {
					OR = {
						rite = rite:armenian_rite
						rite = rite:georgian_rite
						rite = rite:albanian_apostolic
					}
				}

### Create Bolghar culture
create_bolghar_culture_effect = {
	custom_description = {
		text = your_culture_adopts_the_bulghar_identity
		subject = root
		$THE_CULTURE$ = {
			set_culture_name = {
				noun = true_bolghar
				collective_noun = true_bolghar_collective_noun
				prefix = true_bolghar_prefix
			}
			set_name_list = name_list_bolghar
		}
	}
	
	#All your Oghuric Vassals adopt your culture, regardless of their own
	custom_description = {
		text = all_oghuric_vassals_adopt_your_culture
		subject = root
		
		every_vassal_or_below = {
			limit = {
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
			}
			set_culture = $THE_CULTURE$
			every_courtier_or_guest = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
				}
				set_culture = $THE_CULTURE$
			}
		}
		
		every_tributary = {
			limit = {
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
			}
			set_culture = $THE_CULTURE$
			every_courtier_or_guest = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
				}
				set_culture = $THE_CULTURE$
			}
			every_vassal_or_below = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
				}
				set_culture = $THE_CULTURE$
			}
		}
	}
	
	custom_description = {
		text = all_oghuric_vassal_counties_adopt_your_culture
		subject = root
		
		every_realm_county = {
			limit = {
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
			}
			set_county_culture = $THE_CULTURE$
		}
		
		every_tributary = {
			limit = {
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
			}
			every_realm_county = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
				}
				set_county_culture = $THE_CULTURE$
			}
		}
	}
	
	custom_description = {
		text = nearby_oghur_rulers_adopt_your_culture
		subject = root
		
		every_ruler = {
			limit = {
				is_landed = yes
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
				NOT = { root = { target_is_vassal_or_below = prev } }
				NOT = { is_tributary_of = root }
				culture = { general_culture_will_not_flip_culture_trigger = no }
				#Distance Calculation; scope issues making script values difficult
				root = { realm_to_title_distance_squared = { target = prev.primary_title value <= root.prestige_distance_oghuric } }
			}
			set_culture = $THE_CULTURE$
			every_courtier_or_guest = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
					culture = { general_culture_will_not_flip_culture_trigger = no }
				}
				set_culture = $THE_CULTURE$
			}
			every_vassal_or_below = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
					culture = { general_culture_will_not_flip_culture_trigger = no }
				}
				set_culture = $THE_CULTURE$
			}
		}
	}
	
	#Flip Counties
	custom_description = {
		text = nearby_oghur_counties_adopt_your_culture
		subject = root
		
		every_county = {
			limit = {
				is_landless_type_title = no
				this.culture = { has_cultural_pillar = heritage_oghuric }
				NOT = { this.culture = $THE_CULTURE$ }
				NOT = { root = { target_is_vassal_or_below = prev.holder } }
				NOT = { this.holder = { is_tributary_of = root } }
				culture = { general_culture_will_not_flip_culture_trigger = no }
				#Distance Calculation; scope issues making script values difficult
				root = { realm_to_title_distance_squared = { target = prev.primary_title value <= root.prestige_distance_oghuric } }
			}
			set_county_culture = $THE_CULTURE$
		}
	}
	
	hidden_effect = { set_global_variable = bolghar_idenity_formed }
}
