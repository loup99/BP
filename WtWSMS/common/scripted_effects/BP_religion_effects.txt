bp_setup_priest_traits_effect = {
	if = { #Buddhists and Jains are always monastic holders
		limit = {
			rite = { rite_has_parameter = clergy_are_always_monks }
			NOT = { has_trait = devoted }
		}
		add_trait = devoted
	}
	if = { #Sacred Eunuchs
		limit = {
			rite = { rite_has_parameter = clergy_are_always_eunuchs }
			NOT = { has_trait = eunuch }
		}
		add_trait = eunuch_1
	}
	if = { #Scythian Pagan Court Chaplains are Enarei trait
		limit = {
			rite = { rite_has_parameter = utilizes_enarei }
			is_male = yes
			NOT = { has_trait = enarei }
		}
		add_trait = enarei
	}
	if = {
		limit = {
			rite = { rite_has_parameter = clerics_are_vegetarians }
			NOT = { has_trait = vegetarian }
		}
		add_trait = vegetarian
	}
	bp_learn_liturgical_language_effect = yes
}
