# When scoped character is appointed as a minister
got_roman_minister_position_effect = {
	save_scope_as = new_minister

	if = { # If your liege has a ministry, set up titles if needed
		limit = {
			$COUNCILLOR_LIEGE$ = {
				this != scope:new_minister
				bp_has_access_to_roman_ministry_trigger = yes
			}
			NOT = { exists = scope:swapped_position }
		}
		# Grant the ministry title if councillor doesn't have it
		if = {
			limit = {
				NOT = { has_title = $MINISTER_TITLE$ }
			}

			if = {
				limit = {
					$MINISTER_TITLE$ = { is_title_created = no }
				}
				create_title_and_vassal_change = {
					type = created
					save_scope_as = minister_title_creation_change
					add_claim_on_loss = no
				}
				$MINISTER_TITLE$ = {
					change_title_holder = {
						holder = $COUNCILLOR_LIEGE$
						change = scope:minister_title_creation_change
					}
				}
				resolve_title_and_vassal_change = scope:minister_title_creation_change
			}
			create_title_and_vassal_change = {
				type = appointment
				save_scope_as = minister_title_change
				add_claim_on_loss = no
			}

			if = {
				limit = {
					$COUNCILLOR_LIEGE$ = {
						bp_has_access_to_roman_ministry_trigger = yes
					}
				}
				# retire from other ministry primary titles
				scope:new_minister = {
					every_held_title = {
						limit = {
							this != $MINISTER_TITLE$
							bp_title_is_a_roman_ministry_title_trigger = yes
						}
						change_title_holder = {
							holder = current_heir
							change = scope:minister_title_change
						}
					}
				}
			}

			$MINISTER_TITLE$ = {
				change_title_holder = {
					holder = scope:new_minister
					change = scope:minister_title_change
				}
			}
			resolve_title_and_vassal_change = scope:minister_title_change
		}
		# Add the correct title law if it doesn't have it already
		if = {
			limit = {
				exists = $MINISTER_TITLE$.holder
				bp_title_is_supreme_roman_military_title_trigger = yes
				NOT = {
					$MINISTER_TITLE$ = {
						has_title_law = roman_administrative_magister_military_appointment_succession_law
					}
				}
			}
			$MINISTER_TITLE$ = {
				add_title_law = roman_administrative_magister_military_appointment_succession_law
			}
		}
		else_if = {
			limit = {
				exists = $MINISTER_TITLE$.holder
				bp_title_is_supreme_roman_ministry_title_trigger = yes
				NOT = {
					$MINISTER_TITLE$ = {
						has_title_law = roman_administrative_ministry_appointment_succession_law
					}
				}
			}
			$MINISTER_TITLE$ = {
				add_title_law = roman_administrative_ministry_appointment_succession_law
			}
		}
		#ELSE: Other ministries we'd implement

		# Make them a minister if they don't have the council position
		if = {
			limit = {
				NOT = { has_council_position = $MINISTER_POSITION$ }
			}
			top_liege = {
				assign_councillor_type = {
					type = $MINISTER_POSITION$
					target = scope:new_minister
					remove_existing_councillor = yes
				}
			}
		}
		# If we have any vassals remaining, let's move them to where they are supposed to be
		if = {
			limit = {
				vassal_count > 0
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = minister_vassal_change
			}
			every_vassal = {
				change_liege = {
					liege = top_liege # I.e. the top liege
					change = scope:minister_vassal_change
				}
			}
			resolve_title_and_vassal_change = scope:minister_vassal_change
		}
		# If the new minister should be the diarch, make it so
		if = {
			limit = {
				OR = {
					AND = {
						top_liege ?= {
							has_realm_law = roman_chancellor_law
						}
						$MINISTER_TITLE$ = {
							bp_title_is_supreme_roman_ministry_title_trigger = yes
							holder = scope:new_minister
						}
					}
					AND = {
						top_liege ?= {
							has_realm_law = roman_marshal_law
						}
						$MINISTER_TITLE$ = {
							bp_title_is_supreme_roman_military_title_trigger = yes
							holder = scope:new_minister
						}
					}
				}
			}
			top_liege = {
				if = {
					limit = {
						diarch != scope:new_minister
					}
					if = {
						limit = {
							NOT = { has_diarchy_type = roman_admin_premier }
						}
						if = {#  Clear out any old diarchy
							limit = { has_active_diarchy = yes }
							end_diarchy = yes
						}
						try_start_diarchy = roman_admin_premier
					}
					set_diarch = scope:new_minister
				}
			}
		}
		
		# Move capital to realm capital
		$COUNCILLOR_LIEGE$.capital_county = {
			save_scope_as = capital_county
		}
		if = {
			limit = {
				$COUNCILLOR_LIEGE$.capital_county != $MINISTER_TITLE$.title_capital_county
			}
			$MINISTER_TITLE$ = {
				set_capital_county = scope:capital_county
			}
		}
		every_held_title = {
			limit = {
				is_noble_family_title = yes
			}
			set_capital_county = scope:capital_county
		}
		if = {
			limit = { is_ai = yes }
			domicile ?= {
				move_domicile = scope:capital_county.title_province
			}
		}
		if = {
			limit = {
				is_in_army = no
				is_travelling = no
				NOT = { exists = involved_activity }
			}
			set_location_to_default = yes
		}
		
		#Other Effects: Gate domicile building, accomplishments, situation phases,
		#               character modifiers, 
	}
}

destroy_and_fire_roman_minister_effect = {
	save_temporary_scope_as = fired_minister
	$COUNCILLOR_LIEGE$ = {
		hidden_effect = {
			add_treasury = scope:fired_minister.treasury
			destroy_title = $MINISTER_TITLE$
			if = {
				limit = {
					scope:fired_minister = {
						is_councillor_of = $COUNCILLOR_LIEGE$
					}
				}
				$COUNCILLOR_LIEGE$ = {
					always_fire_councillor = scope:fired_minister
				}
			}
		}
	}
}

fired_roman_minister_position_effect = {
	if = {
		limit = {
			$COUNCILLOR_LIEGE$ = {
				bp_has_access_to_roman_ministry_trigger = yes
			}
			NOT = { exists = scope:new_councillor }
		}
		root.primary_title ?= {
			save_scope_as = the_title
			switch = {
				trigger = has_variable
				is_supreme_roman_military_title = {
					if = {
						limit = { exists = scope:the_title.current_heir }
						root.primary_title.current_heir = {
							got_roman_minister_position_effect = {
								MINISTER_TITLE = scope:the_title
								MINISTER_POSITION = councillor_marshal_roman
								COUNCILLOR_LIEGE = $COUNCILLOR_LIEGE$
							}
						}
					}
					else = {
						root = {
							destroy_and_fire_roman_minister_effect = {
								MINISTER_TITLE = scope:the_title
								COUNCILLOR_LIEGE = $COUNCILLOR_LIEGE$
							}
						}
					}
				}
				is_supreme_roman_ministry_title = {
					if = {
						limit = { exists = scope:the_title.current_heir }
						root.primary_title.current_heir = {
							got_roman_minister_position_effect = {
								MINISTER_TITLE = scope:the_title
								MINISTER_POSITION = councillor_chancellor_roman
								COUNCILLOR_LIEGE = $COUNCILLOR_LIEGE$
							}
						}
					}
					else = {
						root = {
							destroy_and_fire_roman_minister_effect = {
								MINISTER_TITLE = scope:the_title
								COUNCILLOR_LIEGE = $COUNCILLOR_LIEGE$
							}
						}
					}
				}
			}
		}
	}
}

setup_magister_military_effect = {
	if = {
	# Ministries enabled
		limit = { BP_has_dlc_necessary_for_roman_ministries_trigger = yes }
		set_variable = is_supreme_roman_military_title
		set_variable = is_a_roman_ministry_title
		if = {
			limit = {
				exists = this.holder
				NOT = { has_title_law = roman_administrative_magister_military_appointment_succession_law }
			}
			add_title_law = roman_administrative_magister_military_appointment_succession_law
			
			# Add to council
			holder ?= {
				prev.hegemony = {
					set_variable = {
						name = councillor_marshal
						value = prev
						days = 1
					}
				}
			}
		}
	}
	# Destroy title without DLC
	else = {
		holder ?= {
			# Clears liquid treasury whenever balance > 0
			empty_treasury_when_abandoning_landed_life_effect = yes
			destroy_title = prev
			
			prev.hegemony = {
				set_variable = {
					name = councillor_marshal
					value = prev
					days = 1
				}
				give_roman_council_position_effect = { POSITION_TYPE = councillor_marshal }
			}
		}
	}
}

# Passed a character from variable after lobby
give_roman_council_position_effect = {

	var:$POSITION_TYPE$ ?= { save_scope_as = councillor }
	
	# Add to court first
	scope:councillor = {
		if = {
			limit = { is_ruler = no }
			set_employer = prev.holder
		}
	}
	holder = {
		if = {
			limit = { bp_has_access_to_roman_ministry_trigger = yes }
			make_councilor_of_type_if_not_already_effect = {
				POSITION = $POSITION_TYPE$_roman
				TARGET_CHAR = scope:councillor
			}
		}
		else = {
			make_councilor_of_type_if_not_already_effect = {
				POSITION = $POSITION_TYPE$
				TARGET_CHAR = scope:councillor
			}
		}
	}
}

establish_premiership_if_necessary_effect = { #Scope is top liege
	if = {
		limit = {
			bp_has_access_to_roman_ministry_trigger = yes
			OR = {
				has_realm_law = roman_chancellor_law
				has_realm_law = roman_marshal_law
			}
			#If we have an active diarchy, it has to be a wrong kind of diarchy.
			#If no active diarchy, always is true
			trigger_if = {
				limit = { has_active_diarchy = yes }
				OR = {
					NOT = { has_diarchy_type = roman_admin_premier }
					AND = {
						has_realm_law = roman_chancellor_law
						this.diarch = { any_held_title = { count < 1 has_variable = is_a_roman_ministry_title } }
					}
					AND = {
						has_realm_law = roman_marshal_law
						this.diarch = { any_held_title = { count < 1 has_variable = is_supreme_roman_military_title } }
					}
				}
			}
			trigger_else = { always = yes }
		}
		# Clear out any old diarchy
		if = {
			limit = { has_active_diarchy = yes }
			end_diarchy = yes
		}
		if = {
			limit = {
				has_realm_law = roman_chancellor_law
				any_vassal = { any_held_title = { has_variable = is_a_roman_ministry_title } }
			}
			try_start_diarchy = roman_admin_premier
			random_vassal = {
				limit = { any_held_title = { has_variable = is_a_roman_ministry_title } }
				prev = { set_diarch = prev }
			}
		}
		else_if = {
			limit = {
				has_realm_law = roman_marshal_law
				any_vassal = { any_held_title = { has_variable = is_supreme_roman_military_title } }
			}
			try_start_diarchy = roman_admin_premier
			random_vassal = {
				limit = { any_held_title = { has_variable = is_supreme_roman_military_title } }
				prev = { set_diarch = prev }
			}
		}
	}
}
