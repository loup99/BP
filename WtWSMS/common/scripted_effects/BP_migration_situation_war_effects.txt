migration_cb_title_and_modifier_application_effect = {
	#Abandon your confederation if you have one, to make things simple
	if = {
		limit = { $ATTACKER$ = { is_confederation_member = yes } }
		random_confederation = {
			limit = { $ATTACKER$ = { is_member_of_confederation = prev } }
			save_scope_as = the_confederation #To add new holders to the old confederation
			remove_confederation_member = $ATTACKER$
		}
	}
	
	create_title_and_vassal_change = {
		type = independency
		save_scope_as = change_attacker_titles_vassals_independent
		add_claim_on_loss = no
	}
	create_title_and_vassal_change = {
		type = swear_fealty
		save_scope_as = change_attacker_titles_vassals_to_liege
		add_claim_on_loss = no
	}
	create_title_and_vassal_change = {
		type = swear_fealty
		save_scope_as = change_defender_vassals_to_attacker
		add_claim_on_loss = no
	}
	create_title_and_vassal_change = {
		type = swear_fealty
		save_scope_as = change_defender_vassals_tributized
		add_claim_on_loss = no
	}
	create_title_and_vassal_change = {
		type = conquest
		save_scope_as = change_conquered_new_lands
		add_claim_on_loss = yes
	}
	
	#### Build necessary lists
	
	#attacker_handed_off_vassals: vassals of attackers who get handed off as vassals
	#attacker_liberated_vassals: vassals of attackers who get liberated
	#attacker_titles_to_independent: Attacker titles going to be independent
	#attacker_titles_to_liege: Attacker titles going to an existing liege
	#defender_titles_taken_in_target: Defender titles taken within the target
	#defender_vassals_in_conquest_region: defending vassals in the conquest region
	#defender_vassals_to_tributize: defending vassals that get liberated
	#defender_vassals_to_transfer: defending vassals that get to become your vassals
	#migrating_attackers: attackers who are migrating
	#migration_source_counties: Counties for which migration modifiers should be applied
	
	random_in_list = {
		list = $TARGET_TITLES$
		save_temporary_scope_as = the_target_title
	}
	
	#If the attacker's primary title is de jure and has no titulars
	#of the same rank, give them one, then destroy the de jure titles
	if = {
		limit = {
			#Have at least one non-titular title equal to your primary tier
			$ATTACKER$ = {
				NOT = {
					any_held_title = {
						count > 0
						is_titular = yes
						this.tier = $ATTACKER$.primary_title.tier
					}
				}
			}
		}
		custom_tooltip = {
			text = you_recieve_a_new_titular_title
			create_title_and_vassal_change = {
				type = created
				save_scope_as = create_a_titular_title
				add_claim_on_loss = yes
			}
			if = {
				limit = { $ATTACKER$.primary_title.tier = tier_duchy }
				create_dynamic_title = {
					tier = duchy
					name = MIGRATORY_TITLE_NAME
					adj = MIGRATORY_TITLE_ADJ
				}
			}
			else_if = {
				limit = { $ATTACKER$.primary_title.tier = tier_kingdom }
				create_dynamic_title = {
					tier = kingdom
					name = MIGRATORY_TITLE_NAME
					adj = MIGRATORY_TITLE_ADJ
				}
			}
			else = {
				create_dynamic_title = {
					tier = duchy
					name = MIGRATORY_TITLE_NAME
					adj = MIGRATORY_TITLE_ADJ
				}
			}
			scope:new_title = {
				change_title_holder = {
					holder = $ATTACKER$
					change = scope:create_a_titular_title
				}
			}
			resolve_title_and_vassal_change = scope:create_a_titular_title
			scope:new_title = {
				set_color_from_title = $ATTACKER$.capital_province.county
				set_capital_county = $ATTACKER$.capital_province.county
				set_coa = $ATTACKER$.house
			}
			$ATTACKER$ = { set_primary_title_to = scope:new_title }
		}
	}
	#Now that you have at least one titular title, destroy every non-titular duchy and above
	$ATTACKER$ = {
		every_held_title = {
			limit = {
				is_titular = no
				this.tier >= tier_duchy
			}
			prev = { destroy_title = prev }
		}
	}
	
	$ATTACKER$ = { #War Attackers and Vassal Handling
		save_temporary_scope_as = war_leader
		add_to_temporary_list = migrating_attackers
		every_vassal = {
			if = {
				limit = { this.top_liege = scope:war_leader }
				add_to_temporary_list = attacker_liberated_vassals
			}
			else = { add_to_temporary_list = attacker_handed_off_vassals }
		}
	}
	
	every_in_list = { #Migration Source Counties
		list = migrating_attackers
		every_held_title = {
			limit = {
				is_landless_type_title = no
				this.tier = tier_county
				this.culture = prev.culture
				this.faith = prev.faith
			}
			add_to_temporary_list = migration_source_counties
		}
	}
	
	every_in_list = { #Attacker title handling
		list = migrating_attackers
		every_held_title = {
			limit = {
				is_landless_type_title = no
				this.tier = tier_county
			}
			if = { #Case liege is migrator
				limit = { this.holder.top_liege = scope:war_leader }
				add_to_temporary_list = attacker_titles_to_independent
			}
			else = { add_to_temporary_list = attacker_titles_to_liege }
		}
	}
	
	every_in_list = { #Defender vassal sorting and title reclamation
		list = $TARGET_TITLES$
		every_in_de_jure_hierarchy = {
			limit = {
				is_landless_type_title = no
				tier = tier_county
				OR = {
					this.holder = $DEFENDER$
					this.holder = { target_is_liege_or_above = $DEFENDER$ }
				}
			}
			add_to_temporary_list = defender_titles_taken
			if = {
				limit = { NOT = { this.holder = $DEFENDER$ } }
				this.holder = { add_to_temporary_list = defender_vassals_in_conquest_region }
			}
		}
	}
	
	every_in_list = { #Defender vassal liberate/subjugate/conquer sorting
		list = defender_vassals_in_conquest_region
		limit = {
			$DEFENDER$ = { government_has_flag = government_is_centralized }
			OR = {
				government_has_flag = government_is_tribal
				government_has_flag = government_is_roman_federate
			}
		}
		if = {
			limit = { this.primary_title.tier < $ATTACKER$.primary_title.tier }
			add_to_temporary_list = defender_vassals_to_transfer
		}
		else_if = {
			limit = {
				OR = {
					this.primary_title.tier >= $ATTACKER$.primary_title.tier
					is_ai = no #Humans get liberated/tributized definitionally
				}
			}
			add_to_temporary_list = defender_vassals_to_tributize
		}
	}
	
	#Remove titles from vassals that survive
	every_in_list = {
		list = defender_titles_taken
		save_temporary_scope_as = the_title_in_question
		every_in_list = {
			list = defender_vassals_to_transfer
			if = {
				limit = { this = scope:the_title_in_question.holder }
				scope:the_title_in_question = { remove_from_list = defender_titles_taken }
			}
		}
		every_in_list = {
			list = defender_vassals_to_tributize
			if = {
				limit = { this = scope:the_title_in_question.holder }
				scope:the_title_in_question = { remove_from_list = defender_titles_taken }
			}
		}
	}
	
	#### Hand off attacker titles
	every_in_list = {
		list = attacker_titles_to_independent
		
		create_character = {
			location = scope:war_leader.location
			culture = this.culture
			faith = this.faith
			gender_female_chance = {
				if = {
					limit = { this.faith = { has_doctrine = doctrine_gender_male_dominated } }
					value = 0
				}
				else_if = {
					limit = { this.faith = { has_doctrine = doctrine_gender_female_dominated } }
					value = 100
				}
				else = { value = 50 }
			}
			trait = peasant_leader
			save_scope_as = new_char
			after_creation = {
				add_trait_xp = {
					trait = peasant_leader
					value = { 10 90 }
				}
			}
		}
		scope:new_char = { add_to_list = new_holders }
		change_title_holder = {
			holder = scope:new_char
			change = scope:change_attacker_titles_vassals_independent
			take_baronies = yes
		}
		scope:new_char = { becomes_independent = { change = scope:change_attacker_titles_vassals_independent } }
	}
	every_in_list = {
		list = attacker_titles_to_liege
		change_title_holder = {
			holder = this.liege
			change = scope:change_attacker_titles_vassals_to_liege
			take_baronies = yes
		}
	}
	
	#### Handle attacker vassals
	every_in_list = {
		list = attacker_liberated_vassals
		becomes_independent = { change = scope:change_attacker_titles_vassals_independent }
	}
	every_in_list = {
		list = attacker_handed_off_vassals
		change_liege = { liege = this.liege.liege change = scope:change_attacker_titles_vassals_to_liege }
	}
	
	#### Take Defender Titles
	every_in_list = {
		list = defender_titles_taken
		change_title_holder = {
			holder = $ATTACKER$
			change = scope:change_conquered_new_lands
			take_baronies = yes
		}
	}
	#### Handle Defender vassals
	every_in_list = {
		list = defender_vassals_to_transfer
		change_liege = { liege = $ATTACKER$ change = scope:change_defender_vassals_to_attacker }
	}
	every_in_list = {
		list = defender_vassals_to_tributize
		becomes_independent = { change = scope:change_defender_vassals_tributized }
		custom_tooltip = {
			text = convert_to_a_tributary
			subject = this
			add_character_flag = default_subjugated_tributary
			BP_break_subject_contract_and_establish_tributary_effect = {
				TRIBUTARY = this
				SUZERAIN = $ATTACKER$
			}
		}
	}
	
	#Record government information for downstream events
	if = {
		limit = { $DEFENDER$ = { government_has_flag = government_is_centralized } }
		$ATTACKER$ = { add_character_flag = centralized_government_bootstrap_available }
	}
	if = {
		limit = { $DEFENDER$ = { government_has_flag = is_a_roman_government } }
		$ATTACKER$ = { add_character_flag = roman_government_bootstrap_available }
		save_sub_roman_type_laws_for_migration_effect = {
			SOURCE_CHAR = $DEFENDER$
			TARGET_CHAR = $ATTACKER$
		}
	}
	
	#Figure out the optimal capital location
	every_in_list = {
		list = defender_titles_taken
		this.duchy = { add_to_list = list_of_duchies }
	}
	ordered_in_list = {
		list = list_of_duchies
		order_by = this.duchy_value_as_capital
		this.title_capital_county = { save_temporary_scope_as = the_new_capital }
	}
	
	#### Update the changes
	resolve_title_and_vassal_change = scope:change_defender_vassals_to_attacker
	resolve_title_and_vassal_change = scope:change_conquered_new_lands
	resolve_title_and_vassal_change = scope:change_attacker_titles_vassals_to_liege
	resolve_title_and_vassal_change = scope:change_attacker_titles_vassals_independent
	resolve_title_and_vassal_change = scope:change_defender_vassals_tributized
	
	#### If you were in a confederation, your old holders join it
	hidden_effect = {
		if = {
			limit = { exists = scope:the_confederation }
			every_in_list = {
				list = attacker_titles_to_independent
				scope:the_confederation = { add_confederation_member = prev.holder }
			}
		}
	}
	
	#### Set your new capital
	$ATTACKER$ = { set_realm_capital = scope:the_new_capital }
	
	#### Disconnected realms become independent
	create_title_and_vassal_change = {
		type = independency
		save_scope_as = change_defender_vassals_exclave_independent
		add_claim_on_loss = no
	}
	$DEFENDER$ = { #Hide this because of tooltips...
		every_vassal = {
			limit = {
				is_landed = yes
				primary_title.tier >= tier_county
				any_held_title = {
					is_landless_type_title = no
					this.tier = tier_county
					NOT = {
						is_connected_to = {
							target = $DEFENDER$.capital_county
							max_naval_distance = 2000
						}
					}
				}
			}
			becomes_independent = { change = scope:change_defender_vassals_exclave_independent }
		}
	}
	resolve_title_and_vassal_change = scope:change_defender_vassals_exclave_independent
	
	custom_description = {
		text = you_abandon_your_current_lands
		object = scope:the_target_title
	}
	
	#### Apply source county modifiers
	custom_description = {
		text = migration_source_counties_get_migrated_away_modifier
		every_in_list = {
			list = migration_source_counties
			if = {
				limit = { has_county_modifier = recent_migration_away_modifier }
				add_county_modifier = recent_migration_away_modifier
			}
			add_county_modifier = { modifier = recent_migration_away_modifier years = 10 }
		}
	}
	
	#### Update County culture/faiths and damage
	custom_description = {
		text = all_new_provinces_get_looted
		object = scope:the_target_title
		$ATTACKER$ = {
			every_realm_province = { #Everything gets looted
				limit = { has_holding = yes }
				if = {
					limit = { has_province_modifier = recently_looted_modifier }
					remove_province_modifier = recently_looted_modifier
				}
				add_province_modifier = {
					modifier = recently_looted_modifier
					years = 5
				}
			}
		}
	}
	custom_description = {
		text = all_conquered_counties_get_recent_migration
		object = scope:the_target_title
		$ATTACKER$ = {
			every_realm_county = {
				limit = { is_target_in_variable_list = { name = migrating_attackers target = this.holder } }
				if = {
					limit = { has_county_modifier = recent_migration_to_modifier }
					remove_county_modifier = recent_migration_to_modifier
				}
				add_county_modifier = {
					modifier = recent_migration_to_modifier
					years = 5
				}
			}
		}
	}
	custom_description = {
		text = counties_may_be_converted_to_tribal_holdings
		object = scope:the_target_title
		$ATTACKER$ = {
			every_realm_county = {
				limit = { NOT = { any_in_list = { list = migrating_attackers this = prev.holder } } }
				migration_flip_holding_to_tribal_effect = {
					THE_ATTACKER = $ATTACKER$
					THE_COUNTY = this
				}
			}
		}
	}
	custom_description = {
		text = all_conquered_counties_may_flip_culture_faith
		object = scope:the_target_title
		migration_flip_culture_faith_in_realm_effect = { THE_ATTACKER = $ATTACKER$ }
	}
}

		}
	}
}

		limit = { has_variable = number_of_migrations_attempted }
		change_variable = { name = number_of_migrations_attempted add = 1 }
	}

migration_flip_holding_to_tribal_effect = {
	#migration_innovation_weight has a max value of 10
	#Development of 10 or all necessary innovations should stop it
	save_temporary_scope_value_as = { name = the_county_development value = $THE_COUNTY$.development_level }
	save_temporary_scope_value_as = { name = the_attacker_culture_weight value = $THE_ATTACKER$.culture.migration_innovation_weight }
	
	random = {
		chance = migration_chance_of_flip_holding
		$THE_COUNTY$ = {
			every_province = {
				limit = {
					is_county_capital = no
					has_province_modifier = recently_looted_modifier
				}
				remove_province_modifier = recently_looted_modifier
			}
			
			raze_county = {
				holding_type = tribal_holding
				purge_secondary_holdings = yes
			}
			$THE_COUNTY$.culture = { save_temporary_scope_as = the_culture }
			$THE_COUNTY$.faith = { save_temporary_scope_as = the_faith }
			
			set_county_culture = $THE_ATTACKER$.culture
			set_county_faith = $THE_ATTACKER$.faith
			
			random_list = {
				60 = {
					add_culture_minority_effect = {
						COUNTY = this
						CULTURE = scope:the_culture
						SIZE = large
					}
					add_faith_minority_effect = {
						COUNTY = this
						FAITH = scope:the_faith
						SIZE = large
					}
				}
				30 = {
					add_culture_minority_effect = {
						COUNTY = this
						CULTURE = scope:the_culture
						SIZE = small
					}
					add_faith_minority_effect = {
						COUNTY = this
						FAITH = scope:the_faith
						SIZE = small
					}
				}
				10 = {}
			}
		}
	}
}
migration_flip_culture_faith_in_realm_effect = {
	#Get the total number of source counties to restrain total number of flips
	$THE_ATTACKER$ = {
		set_variable = {
			name = number_of_migration_source_counties
			value = 0
		}
	}
	every_in_list = {
		list = migration_source_counties
		$THE_ATTACKER$ = {
			change_variable = {
				name = number_of_migration_source_counties
				add = 1
			}
		}
	}
	#Actually apply the flip counties, with the first flip checks at the least developed counties
	$THE_ATTACKER$ = {
		ordered_realm_county = {
			limit = {
				NOT = { this.culture = $THE_ATTACKER$.culture }
				any_in_list = { list = migrating_attackers this = prev.holder }
			}
			order_by = {
				value = this.development_level
				multiply = -1.00
			}
			max = $THE_ATTACKER$.realm_size
			check_range_bounds = no
			
			save_scope_as = the_county
			#Development of 30 should stop it.
			# Each era beyond Tribal reduces odds by 5%
			scope:the_county = { save_temporary_scope_as = the_county }
			scope:the_county.culture = { save_temporary_scope_as = the_county_culture }
			save_temporary_scope_value_as = { name = the_county_development value = scope:the_county.development_level }
			$THE_ATTACKER$ = { save_temporary_scope_as = the_attacker }
			
			random = {
				chance = migration_chance_of_flip_culture_faith
				scope:the_county = {
					scope:the_county.culture = { save_temporary_scope_as = the_culture }
					scope:the_county.faith = { save_temporary_scope_as = the_faith }
					
					set_county_culture = $THE_ATTACKER$.culture
					set_county_faith = $THE_ATTACKER$.faith
					random_list = {
						60 = {
							add_culture_minority_effect = {
								COUNTY = this
