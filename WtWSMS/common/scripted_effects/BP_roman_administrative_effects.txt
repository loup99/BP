setup_magister_military_effect = {
	if = {
	# Ministries enabled
		limit = { has_tgp_dlc_trigger = yes }
		set_variable = is_supreme_roman_military_title
		set_variable = is_a_roman_ministry_title
		if = {
			limit = {
				exists = this.holder
				NOT = { has_title_law = roman_administrative_magister_military_appointment_succession_law }
			}
			add_title_law = roman_administrative_magister_military_appointment_succession_law
			
			# Add to council
			holder ?= {
				vassal_contract_set_obligation_level = {
					type = roman_administrative_themes
					level = roman_admin_theme_ministry_civilian_value
				}
				
				prev.hegemony = {
					set_variable = {
						name = councillor_marshal
						value = prev
						days = 1
					}
				}
			}
		}
	}
	# Destroy title without DLC
	else = {
		holder ?= {
			# Clears liquid treasury whenever balance > 0
			empty_treasury_when_abandoning_landed_life_effect = yes
			destroy_title = prev
			
			prev.hegemony = {
				set_variable = {
					name = councillor_marshal
					value = prev
					days = 1
				}
				give_roman_council_position_effect = { POSITION_TYPE = councillor_marshal }
			}
		}
	}
}

# Passed a character from variable after lobby
give_roman_council_position_effect = {

	var:$POSITION_TYPE$ ?= { save_scope_as = councillor }
	
	# Add to court first
	scope:councillor = {
		if = {
			limit = { is_ruler = no }
			set_employer = prev.holder
		}
	}
	holder = {
		if = {
			limit = { bp_has_access_to_roman_ministry_trigger = yes }
			make_councilor_of_type_if_not_already_effect = {
				POSITION = $POSITION_TYPE$_roman
				TARGET_CHAR = scope:councillor
			}
		}
		else = {
			make_councilor_of_type_if_not_already_effect = {
				POSITION = $POSITION_TYPE$
				TARGET_CHAR = scope:councillor
			}
		}
	}
}

	if = { # If your liege has a ministry, set up titles if needed

			scope:councillor_liege ?= {
				this != scope:new_minister
				bp_has_access_to_roman_ministry_trigger = yes
			}
			NOT = { exists = scope:swapped_position }
		}
		# Grant the ministry title if councilor doesn't have it
		if = {
			limit = {
				NOT = { has_title = $MINISTER_TITLE$ }

			create_title_and_vassal_change = {
				type = appointment

			if = {
				limit = {
					scope:councillor_liege = {
						bp_has_access_to_roman_ministry_trigger = yes
					}
					# retire from other ministry primary titles
					OR = {
						scope:new_minister.primary_title ?= { has_variable = is_supreme_roman_military_title }
						scope:new_minister.primary_title ?= { has_variable = is_supreme_roman_ministry_title }

					save_scope_as = old_ministry_title

						holder = current_heir
						change = scope:minister_title_change
					}
				}
			}

			$MINISTER_TITLE$ = {
				change_title_holder = {

				}

			limit = { scope:new_minister = { is_independent_ruler = yes } }

				type = swear_fealty

				liege = scope:the_top_liege

				trigger = has_government

					if = {
						limit = {
							NOT = { scope:new_minister = { has_government = roman_administrative_government } }
						}
						scope:new_minister = { change_government = roman_administrative_government }
					}
				}

fired_roman_minister_position_effect = {
	if = {
		limit = {
			trigger_if = {
				limit = { exists = scope:councillor_liege }
				scope:councillor_liege = { bp_has_access_to_roman_ministry_trigger = yes }
			}
			trigger_else = { always = yes }
			NOT = { exists = scope:new_councillor }
		}
		root.primary_title ?= {
			save_scope_as = the_title
			switch = {
				trigger = has_variable
				is_supreme_roman_military_title = {
					if = {
						limit = { exists = scope:the_title.current_heir }
						root.primary_title.current_heir = {
							got_roman_administrative_minister_position_effect = {
								MINISTER_TITLE = scope:the_title
								#MINISTER_POSITION = councillor_marshal_roman
							}
						}
					}
					else = {
						root = {
							destroy_and_fire_minister_roman_effect = {
								MINISTER = scope:the_title
							}
						}
					}
				}
				is_supreme_roman_ministry_title = {
					if = {
						limit = { exists = scope:the_title.current_heir }
						root.primary_title.current_heir = {
							got_roman_administrative_minister_position_effect = {
								MINISTER_TITLE = scope:the_title
								#MINISTER_POSITION = councillor_chancellor_roman
							}
						}
					}
					else = {
						root = {
							destroy_and_fire_minister_roman_effect = {
								MINISTER = scope:the_title
							}
						}
					}
				}
			}
		}
	}
}

destroy_and_fire_minister_roman_effect = {
	save_temporary_scope_as = fired_minister
	scope:councillor_liege ?= {
		set_variable = {
			name = can_fire_minister
			value = yes
		}
		hidden_effect = {
			destroy_title = $MINISTER$
			if = {
				limit = {
					scope:fired_minister = {
						is_councillor_of = scope:councillor_liege
					}
				}
				scope:councillor_liege = {
					fire_councillor = scope:fired_minister
				}
			}
		}
		remove_variable = can_fire_minister
	}
}

establish_premiership_if_necessary_effect = { #Scope is top liege
	if = {
		limit = {
			bp_has_access_to_roman_ministry_trigger = yes
			OR = {
				has_realm_law = roman_chancellor_law
				has_realm_law = roman_marshal_law
			}
			#If we have an active diarchy, it has to be a wrong kind of diarchy.
			#If no active diarchy, always is true
			trigger_if = {
				limit = { has_active_diarchy = yes }
				OR = {
					NOT = { has_diarchy_type = roman_admin_premier }
					AND = {
						has_realm_law = roman_chancellor_law
						this.diarch = { any_held_title = { count < 1 has_variable = is_a_roman_ministry_title } }
					}
					AND = {
						has_realm_law = roman_marshal_law
						this.diarch = { any_held_title = { count < 1 has_variable = is_supreme_roman_military_title } }
					}
				}
			}
			trigger_else = { always = yes }
		}
		# Clear out any old diarchy
		if = {
			limit = { has_active_diarchy = yes }
			end_diarchy = yes
		}
		if = {
			limit = {
				has_realm_law = roman_chancellor_law
				any_vassal = { any_held_title = { has_variable = is_a_roman_ministry_title } }
			}
			try_start_diarchy = roman_admin_premier
			random_vassal = {
				limit = { any_held_title = { has_variable = is_a_roman_ministry_title } }
				prev = { set_diarch = prev }
			}
		}
		else_if = {
			limit = {
				has_realm_law = roman_marshal_law
				any_vassal = { any_held_title = { has_variable = is_supreme_roman_military_title } }
			}
			try_start_diarchy = roman_admin_premier
			random_vassal = {
				limit = { any_held_title = { has_variable = is_supreme_roman_military_title } }
				prev = { set_diarch = prev }
			}
		}
	}
}

change_government_subroman_administrative_effect = {
	change_government = subroman_administrative_government
	if = {
		limit = {
			this = this.house.house_head
			OR = {
				this.primary_title.tier >= tier_duchy #Don't want nf running counties...
				#Senators are always noble
				has_trait = senator_west
				has_trait = senator_east
			}
			any_held_title = { count < 1 is_noble_family_title = yes }
		}
		create_noble_family_effect = { GOVERNMENT_GIVER = this }
		#Succession Law update
		if = {
			limit = { is_independent_ruler = yes }
			add_law_if_not_has_it_skip_effects_effect = { LAW = roman_acclamation_succession_law }
		}
		else = {
			add_law_if_not_has_it_skip_effects_effect = { LAW = roman_appointment_succession_law }
		}
		#So domiciles aren't out-of-realm after migration because location hasn't updated...
		custom_tooltip = {
			text = you_get_a_domicile_in_your_capital
			if = {
				limit = { is_landed = yes }
				this.domicile = { move_domicile = prev.capital_county.title_province }
			}
			else = { this.domicile = { move_domicile = prev.top_liege.capital_county.title_province } }
		}
	}
}

#The Actual Hand-off Code
flip_subroman_government_types_effect = { #Root Char 
		#Save Law References
		save_sub_roman_type_law_effect = { SOURCE_TYPE = $SOURCE_TYPE$ }
		#Convert government and to all vassals
		change_government_subroman_$TARGET_TYPE$_effect = yes
		#Random senators get Noble families if house heads
		if = {
			limit = { always = $ENOBLE_COURTIERS$ }
			every_courtier = {
				courtiers_become_admins_effect = {
					TOP_LIEGE = this.court_owner.top_liege
					TYPE = $TARGET_TYPE$
				}
			}
		}
		every_vassal_or_below = {
			limit = { has_government = subroman_$SOURCE_TYPE$_government }
			change_government_subroman_$TARGET_TYPE$_effect = yes
			#Random senators get Noble families if house heads (dynastic2admin)
			if = {
				limit = { always = $ENOBLE_COURTIERS$ }
				every_courtier = {
					if = {
						limit = {
							OR = {
								has_trait = senator_west
								has_trait = senator_east
							}
						}
						set_employer = this.court_owner.top_liege
						courtiers_become_admins_effect = {
							TOP_LIEGE = this.court_owner.top_liege
							TYPE = $TARGET_TYPE$
						}
					}
				}
			}
		}
		#Set Laws accordingly and clear flags
		set_sub_roman_type_law_effect = {
			SOURCE_TYPE = $SOURCE_TYPE$
			TARGET_TYPE = $TARGET_TYPE$
		}
}
courtiers_become_admins_effect = {
	if = {
		limit = {
			this = this.house.house_head
			OR = {
				#Senators are always noble
				has_trait = senator_west
				has_trait = senator_east
			}
			any_held_title = { count < 1 is_noble_family_title = yes }
		}
		create_noble_family_effect = { GOVERNMENT_GIVER = $TOP_LIEGE$ }
		change_government = subroman_$TYPE$_government
	}
}
