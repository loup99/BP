paganify_easter_egg_effect = {
	#Adjust ethnic religions split across heritages
	religion:proto_carpathian_religion = {
		every_faith = {
			swap_doctrine_if_it_has_it_effect = {
				OLD_DOCTRINE = tenet_ethnic_religion
				NEW_DOCTRINE = tenet_communal_identity
			}
		}
	}
	
	#Convert counties and characters
	every_county = {
		limit = { exists = county }
		paganify_county_effect = yes
	}
	every_ruler = {
		limit = {
			is_from_ruler_designer = no
			NOT = { faith.religious_head = this }
		}
		paganify_ruler_effect = yes
	}
}

paganify_county_effect = {
	get_paganify_faith_effect = yes
	if = {
		limit = { exists = this.var:paganify_faith }
		
		# Add minority on holy site
		if = {
			limit = { NOT = { any_county_province = { barony = { is_holy_site_of = prev.faith } } } }
			remove_list_variable = {
				name = faith_minorities_small
				target = faith
			}
			remove_list_variable = {
				name = faith_minorities_large
				target = faith
			}
		}
		
		set_county_faith = this.var:paganify_faith
		remove_list_variable = {
			name = faith_minorities_small
			target = this.var:paganify_faith
		}
		remove_list_variable = {
			name = faith_minorities_large
			target = this.var:paganify_faith
		}
	}
}

paganify_ruler_effect = {
	get_paganify_faith_effect = yes
		
	every_courtier_or_guest = {
		if = {
			limit = {
				exists = prev.var:paganify_faith
				OR = {
					this.faith = prev.faith
					this.culture = { has_same_culture_heritage = prev.culture }
					has_council_position = councillor_court_chaplain
				}
			}
			set_character_faith = prev.var:paganify_faith
		}
		else = {
			get_paganify_faith_effect = yes
			if = {
				limit = { exists = this.var:paganify_faith }
				set_character_faith = this.var:paganify_faith
			}
		}
	}
	# Set own faith
	if = {
		limit = { exists = this.var:paganify_faith }
		set_character_faith = this.var:paganify_faith
	}
}
