#Macros
set_variable_on_root_culture_effect = { #NEW_CULTURE scope
	if = {
		limit = { has_variable = root_culture }
		this.var:root_culture ?= { set_variable = { name = $NAME$ value = $VALUE$ } }
	}
	#Inform devs of failures
	else = { debug_log = "ERROR: DIVERGENT NAME TRACKING LACKS A ROOT" }
}
add_variable_to_title_list_on_root_culture_effect = { #NEW_CULTURE scope
	if = {
		limit = { has_variable = root_culture }
		this.var:root_culture ?= {
			add_to_variable_list = {
				name = $NAME$
				target = $VALUE$
			}
		}
		
	}
	#Inform devs of failures
	else = { debug_log = "ERROR: DIVERGENT NAME TRACKING LACKS A ROOT" }
}
#Variables to track: title_culture_name, root_culture, migration_culture_divergence_*, high_culture
pick_a_dev_specified_name_if_possible_effect = {
	#Any and all !very special! dev divergence names get priority here,
	#including clearing out history variables if need be
	$CULTURE_FORMER$ = {}
	$OLD_CULTURE$ = {}
	
	#Historical Examples:
	# Przeworsk, Wielbark, Alauni, Halani would all need very custom naming handling here,
	# with priority over anything else we might pick.
	#
	#Potential Mod Examples:
	# Alamanni becoming Swabians if creation names fail and no Suebi left as non-minorities?
	# Baivarii ethnogenesis with Rygir migration?
	
	set_variable = no_custom_name_picked
}

pick_a_title_name_if_possible_effect = {
	if = { #Titular Logic
		limit = { $CULTURE_FORMER$.primary_title ?= { is_titular = yes } }
		$CULTURE_FORMER$ = { save_temporary_scope_as = the_culture_former_reference }
		set_culture_name = {
			noun = migration_divergent_culture_titular_title_noun
			collective_noun = migration_divergent_culture_titular_title_collective_noun
			prefix = migration_divergent_culture_titular_title_prefix
		}
	}
	#Non-Titular special title-related cases; should be very culture dependent
	else_if = { #Franks
		limit = {
			OR = {
				$CULTURE_FORMER$.culture = culture:old_frankish
				$CULTURE_FORMER$.culture.var:root_culture ?= culture:old_frankish
			}
		}
		if = { #Salian Franks
			limit = { $CULTURE_FORMER$.primary_title ?= title:d_belgica_secunda }
			set_culture_name = {
				noun = migration_divergent_culture_salian_franks_noun
				collective_noun = migration_divergent_culture_salian_franks_collective_noun
				prefix = migration_divergent_culture_salian_franks_prefix
			}
		}
		else_if = { #Ripuarian Franks
			limit = { $CULTURE_FORMER$.primary_title ?= title:d_germania_inferior }
			set_culture_name = {
				noun = migration_divergent_culture_ripuarian_franks_noun
				collective_noun = migration_divergent_culture_ripuarian_franks_collective_noun
				prefix = migration_divergent_culture_ripuarian_franks_prefix
			}
		}
		else_if = { #Texandrian Franks
			limit = { $CULTURE_FORMER$.primary_title ?= title:d_texandria }
			set_culture_name = {
				noun = migration_divergent_culture_texandrian_franks_noun
				collective_noun = migration_divergent_culture_texandrian_franks_collective_noun
				prefix = migration_divergent_culture_texandrian_franks_prefix
			}
		}
		else_if = { #Mosanian Franks
			limit = { $CULTURE_FORMER$.primary_title ?= title:d_mosania }
			set_culture_name = {
				noun = migration_divergent_culture_mosanian_franks_noun
				collective_noun = migration_divergent_culture_mosanian_franks_collective_noun
				prefix = migration_divergent_culture_mosanian_franks_prefix
			}
		}
		$CULTURE_FORMER$.primary_title = { save_scope_as = the_primary_title }
		add_variable_to_title_list_on_root_culture_effect = {
			NAME = title_culture_name
			VALUE = scope:the_primary_title
		}
	}
	#Fallback
	else = { set_variable = no_custom_name_picked }
}
make_high_culture_if_possible_effect = {
	if = {
		limit = {
			#Either the triggering character is driving the event, or the former has a higher title rank
			OR = {
				$EVENT_TRIGGER_CHAR$ = $CULTURE_FORMER$
				$CULTURE_FORMER$.primary_title.tier > $EVENT_TRIGGER_CHAR$.primary_title.tier
			}
			
		}
		set_culture_name = {
			noun = migration_divergent_culture_high_culture_noun
			collective_noun = migration_divergent_culture_high_culture_collective_noun
			prefix = migration_divergent_culture_high_culture_prefix
		}
		set_variable_on_root_culture_effect = {
			NAME = high_culture
			VALUE = yes
		}
	}
	#Fallback
	else = { set_variable = no_custom_name_picked }
}
make_culture_old_if_possible_effect = {
	set_culture_name = {
		noun = migration_divergent_culture_old_noun
		collective_noun = migration_divergent_culture_old_collective_noun
		prefix = migration_divergent_culture_old_prefix
	}
	set_variable_on_root_culture_effect = {
		NAME = old_culture
		VALUE = yes
	}
}
pick_a_new_culture_name_effect = {
	#EVENT_TRIGGER_CHAR - The Char triggering the divergence event
	#CULTURE_FORMER - Char diverging new culture
	#NEW_CULTURE - Newly Created Culture
	#OLD_CULTURE - Parent Culture
	
	#scope:old_culture must be set to have title strings trigger appropriately
	$OLD_CULTURE$ = { save_scope_as = old_culture }
	
	#Pick a regional name if you can; this is a walk-through set checking if no_custom_name_picked
	#at each stage. and when it is no longer set, it'll skip the rest of the chain
	$NEW_CULTURE$ = {
		pick_a_dev_specified_name_if_possible_effect = {
			CULTURE_FORMER = $CULTURE_FORMER$
			OLD_CULTURE = scope:old_culture
		}
		if = { #Pick a regional name if possible; variable existence is checked on a per-variable level
			limit = { has_variable = no_custom_name_picked }
			remove_variable = no_custom_name_picked
			pick_a_regional_name_if_possible_effect = {
				CULTURE_FORMER = $CULTURE_FORMER$
			}
		}
		if = { #Pick a title name if possible
			limit = {
				has_variable = no_custom_name_picked
				#Wrap this level with a general check for variable list
				trigger_if = {
					limit = { this.var:root_culture ?= { has_variable_list = title_culture_name } }
					NOT = {
						this.var:root_culture ?= {
							is_target_in_variable_list = {
								name = title_culture_name
								target = $CULTURE_FORMER$.primary_title
							}
						}
					}
				}
			}
			remove_variable = no_custom_name_picked
			pick_a_title_name_if_possible_effect = {
				CULTURE_FORMER = $CULTURE_FORMER$
			}
		}
		if = { #Create "High" <CULTURE> if possible
			limit = {
				has_variable = no_custom_name_picked
				NOT = { this.var:root_culture ?= { has_variable = high_culture } }
			}
			remove_variable = no_custom_name_picked
			make_high_culture_if_possible_effect = {
				CULTURE_FORMER = $CULTURE_FORMER$
				EVENT_TRIGGER_CHAR = $EVENT_TRIGGER_CHAR$
			}
			#Special cultural title case ("High <CULTURE>") (TODO LOGIC)
		}
		if = { #Create "Old" <CULTURE> if possible
			limit = {
				has_variable = no_custom_name_picked
				NOT = { this.var:root_culture ?= { has_variable = old_culture } }
			}
			remove_variable = no_custom_name_picked
			make_culture_old_if_possible_effect = yes
		}
		#If we get here, just... go with whatever CK3 cooked up
	}
	#Garbage collection
	if = {
		limit = { $NEW_CULTURE$ = { has_variable = no_custom_name_picked } }
		$NEW_CULTURE$ = { remove_variable = no_custom_name_picked }
	}
}
