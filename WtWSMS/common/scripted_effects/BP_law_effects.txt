clerical_authory_angry_at_reduced_clerical_influence_effect = {
	#All priests are pissed you lowered their influence
	every_vassal = {
		limit = {
			OR = {
				government_has_flag = government_is_theocracy
				has_vassal_stance = zealot
			}
			faith = {
				this.religion = $TOP_LIEGE$.religion
				this.religious_head = $TOP_LIEGE$.faith.religious_head
			}
		}
		if = {
			limit = { this = $TOP_LIEGE$.faith.religious_head }
			add_opinion = {
				modifier = hof_vassal_reduced_influence_modifier
				target = root
				years = 25
			}
		}
		else_if = {
			add_opinion = {
				modifier = clerical_vassal_reduced_influence_modifier
				target = root
				years = 25
			}
		}
		else = { #Secular Zealots
			add_opinion = {
				modifier = zealot_vassal_reduced_clerical_influence_modifier
				target = root
				years = 25
			}
		}
		every_court_position_holder = { #And update the landless realm priest/devoted opinions
			limit = {
				OR = {
					has_council_position = councillor_court_chaplain
					has_trait = devoted
				}
				is_landed = no
			}
			add_opinion = {
				modifier = clerical_vassal_reduced_influence_modifier
				target = root
				years = 25
			}
		}
	}
}

subroman_TYPE_aristocracy_law_can_pass_effect = {
	if = {
		limit = { top_liege = this }
		set_variable = {
			name = subroman_$TYPE$_aristocracy_law_cooldown
			years = subroman_$TYPE$_aristocracy_law_cooldown
		}
		if = {
			limit = {
				has_realm_law = subroman_$TYPE$_aristocracy_law_$LESS_ARISTOCRATIC_LEVEL$
				 NOT = { $LESS_ARISTOCRATIC_LEVEL$ = $CURRENT_ARISTOCRATIC_LEVEL$ }
			}
			add_prestige = 150
			if = {
				limit = { always = $USES_INFLUENCE$ }
				change_influence = 150
			}
			every_vassal = { #Improve Law One Level
				limit = {
					bp_realm_law_use_subroman_$TYPE$_aristocracy_trigger = yes
					NOT = { has_realm_law = subroman_$TYPE$_aristocracy_law_$MORE_ARISTOCRATIC_LEVEL$ }
				}
				random = {
					chance = subroman_chance_vassal_gets_aristocratic_law_improve_one_level
					improve_aristocracy_law_one_level_effect = yes
					send_interface_message = {
						type = msg_liege_passed_law
						title = top_liege_makes_aristocracy_more_powerful_tt
						desc = top_liege_makes_aristocracy_more_powerful_desc
					}
				}
			}
		}
		else_if = {
			limit = {
				has_realm_law = subroman_$TYPE$_aristocracy_law_$MORE_ARISTOCRATIC_LEVEL$
				 NOT = { $CURRENT_ARISTOCRATIC_LEVEL$ = $MORE_ARISTOCRATIC_LEVEL$ }
			}
			every_vassal = { #Decline Law One Level
				limit = {
					bp_realm_law_use_subroman_$TYPE$_aristocracy_trigger = yes
					NOT = { has_realm_law = subroman_$TYPE$_aristocracy_law_$LEAST_ARISTOCRATIC_LEVEL$ }
					NOT = { has_realm_law = subroman_$TYPE$_aristocracy_law_$MOST_ARISTOCRATIC_LEVEL$ }
				}
				random = {
					chance = subroman_chance_vassal_gets_aristocratic_law_decline_one_level
					decline_aristocracy_law_one_level_effect = yes
					send_interface_message = {
						type = msg_liege_passed_law
						title = top_liege_makes_aristocracy_less_powerful_tt
						desc = top_liege_makes_aristocracy_less_powerful_desc
					}
				}
			}
		}
	}
}
