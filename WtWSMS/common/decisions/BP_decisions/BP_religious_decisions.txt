### Christian Decisions
raise_circumcellions_decision = {
	picture = {
		reference = "gfx/interface/illustrations/tournament_contest_illustrations/horse_racing_success.dds"
	}
	decision_group_type = religious

	desc = raise_circumcellions_decision_desc
	selection_tooltip = raise_circumcellions_decision_tooltip
	
	is_shown = {
		is_ruler = yes
		rite = { rite_has_parameter = raise_circumcellions_decision_visible }
		NOT = { government_has_flag = government_is_nomadic }
		trigger_if = { #Available for high tribals, but not tribals
			limit = { government_has_flag = government_is_tribal }
			government_has_flag = government_is_settled
		}
	}
	
	is_valid = {
		has_personal_tenet_flag = raise_circumcellions_decision
		trigger_if = {
			limit = { has_character_flag = recently_called_circumcellions }
			custom_description = {
				text = recently_called_circumcellions
				NOT = { has_character_flag = recently_called_circumcellions }
			}
		}
		custom_tooltip = {
			text = at_least_on_county_is_your_rite
			subject = root
			any_sub_realm_county = { this.rite = root.rite }
		}
		piety_level >= 1
		is_at_war = yes
		#Must be at war with an infidel as the primary
		custom_description = { #Change to faith hostility level must be Evil
			text = must_be_fighting_infidel_as_primary_opponent
			any_character_war = {
				is_war_leader = root
				save_temporary_scope_as = the_war_in_question
				root = {
					any_war_enemy = {
						scope:the_war_in_question = { is_war_leader = root }
						save_temporary_scope_as = the_other_leader
						root.rite = { rite_hostility_level = { target = scope:the_other_leader.rite value > faith_astray_level } }
					}
				}
			}
		}
		NOT = { has_trait = excommunicated } #Definitely not allowed in this case; can't fight the Bishop of Carthage deposing you
	}
	
	is_valid_showing_failures_only = {
		completely_controls = root.capital_county
		piety >= 0
		is_imprisoned = no
		is_available_adult = yes
	}
	
	cost = { piety = root.circumellion_piety_cost }
	
	effect = {
		if = {
			limit = { NOT = { any_sub_realm_county = { this.rite = root.rite } } }
			custom_tooltip = {
				text = no_effect
				subject = root
			}
		}
		else = {
			root = {
				save_temporary_scope_as = the_ruler
				ordered_character_war = {
					order_by = given_war_other_side_max_military_strength
					position = 0
					save_scope_as = chosen_war
				}
			}
			every_realm_county = {
				limit = {
					this.holder = root
					this.rite = root.rite
				}
				root = {
					spawn_army = {
						levies = 600
						men_at_arms = {
							type = light_footmen
							stacks = 3
						}
						location = scope:the_ruler.capital_province
						uses_supply = no
						war = scope:chosen_war
						inheritable = no
						name = circumcellions_army_name
					}
				}
				#The peasants are ~~revolting~~ fighting for us, sir.
				change_county_control = -25
				#And the impact sticks around for years
				add_county_modifier = { modifier = county_raised_circumcellions years = 5 }
			}
			hidden_effect = {
				root = { add_character_flag = { flag = recently_called_circumcellions years = 5 } }
				root = { save_temporary_scope_as = circumellions_raiser }
				every_character_war = {
					every_war_participant = {
						#limit = { NOT = { this = root } }
						send_interface_message = {
							type = event_generic_neutral
							title = circumellions_raised_title
							desc = circumellions_raised_desc
							tooltip = circumellions_raised_tt
							left_icon = root
						}
					}
				}
			}
		}
	}
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 12
		duchy = 24
		kingdom = 48
		empire = 96
		hegemony = 120
	}
	
	ai_potential = { is_at_war = yes }
	
	ai_will_do = {
		base = 50 # 50% Chance in general (we want it to be likely to happen if conditions are right)
		#Zealous and Ambitious required, also needs enough piety
		
		#Won't do if they can't pay
		modifier = {
			factor = 0
			NOT = { piety >= root.circumellion_piety_cost }
		}
		#Won't do if cynical
		modifier = {
			factor = 0
			has_trait = cynical
		}
		#Won't do if they aren't significantly outgunned
		modifier = {
			factor = 0
			root.my_side_max_military_strength > root.three_fourths_other_sides_max_military_strength
		}
		#Arrogant halves the chance (Having the peasants help us?!)
		modifier = {
			factor = 0.5
			has_trait = arrogant
		}
		#Likely to do if Zealous/Ambitious (it'll keep them in the Empire and all that)
		#Will do if both zealous and ambitious, even if arrogant
		modifier = {
			factor = 1.59 #sqrt(2.5) rounded up
			has_trait = zealous
		}
		modifier = {
			factor = 1.59 #sqrt(2.5) rounded up
			has_trait = ambitious
		}
	}
}
reaffirm_faith_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}
	decision_group_type = religious

	desc = reaffirm_faith_decision_desc
	selection_tooltip = reaffirm_faith_decision_tooltip
	
	is_shown = {
		OR = {
			rite = { rite_has_parameter = reaffirm_faith_decision_visible }
			has_personal_tenet_flag = reaffirm_faith_decision
		}
	}
	
	is_valid = {
		has_personal_tenet_flag = reaffirm_faith_decision
		NOT = { has_trait = excommunicated } #Definitely not allowed in this case; get your affairs in order first
		NOT = { has_trait = cynical }
		trigger_if = {
			limit = { has_character_flag = recently_reaffirmed_faith }
			custom_description = {
				text = recently_reaffirmed_faith
				NOT = { has_character_flag = recently_reaffirmed_faith }
			}
		}
	}
	
	is_valid_showing_failures_only = {
		piety >= 0
		is_imprisoned = no
		is_available_adult = yes
	}
	
	cost = {
		piety = {
			value = 50
			multiply = primary_title.tier
			if = {
				limit = { has_trait = zealous }
				multiply = 0.5
			}
		}
	}
	
	effect = {
		hidden_effect = {
			root = { add_character_flag = { flag = recently_reaffirmed_faith years = 5 } }
		}
		change_spiritual_fulfillment = {
			value = 10
			if = {
				limit = { root = { has_trait = fickle } }
				add = 2
			}
			if = {
				limit = { root = { has_trait = eccentric } }
				add = 2
			}
			every_character_trait = {
				save_temporary_scope_as = the_trait
				if = {
					limit = { root.rite = { trait_is_sin_rite = scope:the_trait } }
					add = {
						value = -5
						if = {
							limit = { root = { has_trait = zealous } }
							subtract = 1
						}
						subtract = {
							value = 0
							if = {
								limit = { root.piety_level >= 1 }
								subtract = 1
							}
							if = {
								limit = { root.piety_level >= 2 }
								subtract = 2
							}
							if = {
								limit = { root.piety_level >= 3 }
								subtract = 3
							}
							if = {
								limit = { root.piety_level >= 4 }
								subtract = 4
							}
							if = {
								limit = { root.piety_level >= 5 }
								subtract = 5
							}
						}
					}
				}
				else_if = {
					limit = { root.rite = { trait_is_virtue_rite = scope:the_trait } }
					add = {
						value = 5
						if = {
							limit = { root = { has_trait = zealous } }
							subtract = 1
						}
						subtract = {
							value = 0
							if = {
								limit = { root.piety_level >= 1 }
								subtract = 5
							}
							if = {
								limit = { root.piety_level >= 2 }
								subtract = 4
							}
							if = {
								limit = { root.piety_level >= 3 }
								subtract = 3
							}
							if = {
								limit = { root.piety_level >= 4 }
								subtract = 2
							}
							if = {
								limit = { root.piety_level >= 5 }
								subtract = 1
							}
						}
					}
				}
			}
			min = 5
		}
		custom_tooltip = reaffirm_faith_factors
	}
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 120
		duchy = 120
		kingdom = 120
		empire = 120
		hegemony = 120
	}
	
	ai_potential = { is_at_war = no }
	
	ai_will_do = {
		base = 100
	}
}

### Pagan Decisions
automutiliation_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_physician.dds"
	}
	decision_group_type = religious

	desc = automutiliation_decision_desc_alt
	selection_tooltip = automutiliation_decision_tooltip

	is_shown = {
		is_male = yes
		rite = { rite_has_parameter = show_automutiliation_decision }
	}

	is_valid = {
		NOT = { has_trait = eunuch_1 }
		OR = {
			trigger_if = {
				limit = { rite = { rite_has_parameter = zealous_character_automutilation } }
				has_trait = zealous
			}
			trigger_if = {
				limit = { has_personal_tenet_flag = unlocks_automutiliation_decision }
				NOT = { has_trait = cynical }
			}
		}
		#You can't hurt yourself without at least an heir
		trigger_if = {
			limit = { is_ruler = yes }
			custom_description = {
				text = text_is_a_ruler_and_has_a_player_heir
				exists = player_heir
			}
		}
	}

	effect = {
		add_piety_level = 1
		change_spiritual_fulfillment = 20
		add_trait = devoted
		add_trait = eunuch_1
		
		#Remove spouses and spouse adjacent
		save_scope_as = actor
		if = {
			limit = { is_married = yes }
			every_spouse = { divorce = scope:actor }
		}
		if = {
			limit = { exists = betrothed }
			break_betrothal = betrothed
		}
		if = {
			limit = { is_concubine = yes }
			this.concubinist = { remove_concubine = scope:actor }
		}
		if = {
			limit = { number_of_concubines > 0 }
			every_concubine = { scope:actor = { remove_concubine = prev } }
		}
		show_as_tooltip = {
			random_list = {
				90 = {}
				4 = { add_trait = ill }
				3 = { add_trait = wounded_1 }
				2 = { add_trait = wounded_2 }
				1 = { add_trait = wounded_3 }
			}
		}
	}
	
	ai_check_interval = 60
	
	ai_potential = {
		exists = primary_heir
		primary_heir = { dynasty = root.dynasty }
		is_at_war = no
	}

	#Cribbed from Consolateum
	ai_will_do = {
		base = 0
		
		# 20% chance at 100 zeal
		ai_value_modifier = {
			ai_zeal = 0.2
		}
		
		modifier = {
			add = 5
			is_close_family_of = primary_heir
		}
	}
}
