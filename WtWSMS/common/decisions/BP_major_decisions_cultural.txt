create_ostrogothic_culture_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_varangian_guard.dds"
	}
	decision_group_type = major
	
	title = create_ostrogothic_culture_decision_title
	desc = create_ostrogothic_culture_decision_desc
	selection_tooltip = create_ostrogothic_culture_decision_tooltip
	sort_order = 5000

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		root.culture = culture:gothic
		is_ruler = yes
		NOT = { has_global_variable = ostrogothic_culture_formed }
	}

	is_valid = {
		custom_description = {
			text = gothic_culture_control_all_counties
			subject = root
			object = culture:gothic
			culture:gothic = {
				any_culture_county = {
					OR = {
					this.holder.top_liege = root
					this.holder = root
					}
					percent = 1.00
				}
			}
		}
		#You are the culture head (redundent in 1.14.2, but might not be in next version)
		custom_description = {
			text = is_the_gothic_culture_head
			subject = culture:gothic
			culture:gothic.culture_head = root
		}
		#You are of a sufficient rank
		root.primary_title.tier >= tier_duchy
	}
	
	is_valid_showing_failures_only = {
		is_available_at_peace_adult = yes
		is_imprisoned = no
		is_adult = yes
	}

	cost = {
		prestige = 150
	}

	effect = {
		save_scope_as = culture_former
		show_as_tooltip = { create_ostrogothic_culture_effect = { CREATING_CHAR = root } }
		trigger_event = BP_major_culture_decisions.0001
	}

	#This is mostly to make it rare for the AI to actually attempt
	ai_potential = {
		prestige > 150
	}

	ai_will_do = {
		base = 100
	}
}

adopt_bolghar_identity_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/kubrat_and_his_sons.dds"
	}
	decision_group_type = major

	desc = adopt_bolghar_identity_decision_desc
	selection_tooltip = adopt_bolghar_identity_decision_tooltip

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		#Oghuric Culture in some form
		culture = { has_cultural_pillar = heritage_oghuric }
		is_landed = yes
		NOT = { has_global_variable = bolghar_idenity_formed }
	}

	is_valid = {
		#Must be at least a duke
		custom_description = {
			text = you_must_be_a_duke
			subject = root
			primary_title.tier >= tier_duchy
		}
		#Be noticeably famous
		OR = {
			AND = {
				prestige_level >= 3
				trigger_if = {
					limit = { has_legitimacy = yes }
					legitimacy_level >=3
				}
			}
			has_trait = legendary_leader
		}
		#Be culture head
		custom_tooltip = {
			text = you_are_the_culture_head
			root = culture.culture_head
		}
		#Have Oral Law and Social Stratification
		culture = { has_innovation = innovation_leges_barbarorum }
		culture = { has_innovation = innovation_social_stratification }
		#Control 75% of your culture
		custom_description = {
			text = you_control_at_least_75_percent_of_your_culture_counties
			subject = root
			object = root.culture
			root.percent_of_self_culture_counties_owned > 0.75
		}
		#Control >50% of all Oghuric cultures
		custom_description = {
			text = you_control_at_least_50_percent_of_oghur_culture_counties
			root.percent_of_oghur_heritage_counties_owned > 0.50
		}
	}
	
	cost = {
		prestige = 50 #C'mon, you gotta spend some thing to pull people together...
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		is_available_adult = yes
	}

	effect = {
		save_scope_as = culture_former
		show_as_tooltip = { create_bolghar_culture_effect = { THE_CULTURE = root.culture } }
		hidden_effect = { trigger_event = BP_major_culture_decisions.0002 }
	}
	
	#AI will always do this if possible
	ai_potential = { always = yes }
	
	ai_will_do = {
		base = 100
	}
}

2000th_anniversery_of_gothic_kingdom_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/kubrat_and_his_sons.dds"
	}
	decision_group_type = major

	desc = 2000th_anniversery_of_gothic_kingdom_decision_desc
	selection_tooltip = 2000th_anniversery_of_gothic_kingdom_decision_tooltip

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 120
		hegemony = 60
	}

	is_shown = {
		culture = { any_parent_culture_or_above = { this = culture:gothic } }
		culture = { has_cultural_pillar = heritage_gothic }
		NOT = { has_global_variable = 2000th_anniversery_of_gothic_kingdom }
	}

	is_valid = {
		#Have to be the dynast
		this = this.dynasty.dynast
		#Have Oral Law, Social Stratification, and Formalized Rituals
		culture = { has_innovation = innovation_leges_barbarorum }
		culture = { has_innovation = innovation_social_stratification }
		culture = { has_innovation = innovation_formalized_rituals }
		#Must be an Emperor or Hegemon
		primary_title.tier >= tier_empire
		#Have to have an impressive Realm
		realm_size >= 150
		OR = {
			AND = {
				prestige_level >= 3
				has_trait = legendary_leader
			}
			AND = {
				prestige_level >= 4
				trigger_if = {
					limit = { has_legitimacy = yes }
					legitimacy_level >= 5
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		is_available_adult = yes
	}

	cost = {
		prestige = 1000
		piety = 100
		
		treasury = {
			value = 0
			if = {
				limit = { has_treasury = yes }
				add = {
					value = 50
					every_realm_county = { add = 5 }
				}
				multiply = {
					value = 10
					subtract = this.dynasty_prestige_level
					max = 5
				}
			}
		}
		gold = {
			value = 0
			if = {
				limit = { has_treasury = no }
				add = {
					value = 50
					every_realm_county = { add = 5 }
				}
				multiply = {
					value = 10
					subtract = dynasty_prestige_level
					max = 5
				}
			}
		}
	}

	effect = {
		set_global_variable = 2000th_anniversery_of_gothic_kingdom
		save_scope_as = the_celebrant
		show_as_tooltip = { celebrate_2000_year_anniversery_of_goths_effect = yes }
		hidden_effect = { trigger_event = { id = BP_major_culture_decisions.0100 } }
	}
	
	#AI will always do this if possible
	ai_potential = { always = yes }
	
	ai_will_do = {
		base = 100
	}
}
