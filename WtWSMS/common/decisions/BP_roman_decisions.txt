### Major Decisions
ask_for_hre = {
	picture = {
		reference = "gfx/interface/illustrations/activity_header_backgrounds/activity_coronation.dds"
	}
	decision_group_type = roman
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 60
		hegemony = 0
	}
	
	desc = ask_for_hre_desc
	selection_tooltip = ask_for_hre_tooltip
	
	is_shown = {
		#Haven't already asked
		root = { NOT = { has_variable = asked_for_hre } }
		#West Rome must not exist
		NOT = { exists = title:h_western_roman_empire.holder }
		NOT = { exists = title:e_wre.holder }
		NOT = { exists = title:e_hre.holder }
		#Have to be in communion with the Pope to do this
		faith = {
			OR = {
				this = faith:nicene
				this.religious_head = faith:nicene.religious_head
			}
		}
		age >= 16
		primary_title.tier = tier_empire
		#Only Non-Romans would dare broach this subject this way
		culture = { has_ability_to_reform_rome_directly_trigger = no }
		#Can't be of a specially excluded title
		NOR = {
			#Can't already be Roman Empire
			has_title = title:e_wre
			has_title = title:e_byzantium
			has_title = title:h_western_roman_empire
			has_title = title:h_eastern_roman_empire
			has_title = title:h_roman_empire
			#Axum is special
			has_title = title:e_axum
		}
	}
	
	is_valid = {
		custom_tooltip = {
			text = primary_title_must_be_empire
			primary_title.tier = tier_empire
		}
		realm_size >= 255
		custom_description = {
			text = must_control_at_least_33_percent_of_wre_territory
			subject = root
			title:h_western_roman_empire = {
				any_county = {
					percent >= 0.33
					this.holder.top_liege = root
				}
			}
		}
		OR = {
			has_trait = legendary_leader
			AND = {
				prestige_level >= 5
				legitimacy_level >= 5
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}
	
	cost = {
		treasury = {
			value = 0
			if = {
				limit = { has_treasury = yes }
				add = massive_gold_value
			}
		}
		gold = {
			value = 0
			if = {
				limit = { has_treasury = no }
				add = massive_gold_value
			}
		}
	}
	
	effect = {
		root = {
			set_variable = { name = asked_for_hre }
			save_scope_as = empire_petitoner
		}
		this.faith.religious_head = { trigger_event = { id = BP_roman_decisions.0001 days = { 5 17 } } }
	}
	
	ai_potential = { always = yes }
	
	ai_will_do = {
		base = 10 #%10 chance
		#Zealous and Ambitious required, also needs enough gold
		
		#Won't do without enough gold
		modifier = {
			factor = 0
			NOT = {
				OR = {
					gold >= massive_gold_value
					treasury >= massive_gold_value
				}
			}
		}
		modifier = {
			factor = 2.5
			has_trait = cynical
		}
		modifier = {
			factor = 2.5
			has_trait = ambitious
		}
	}
}

			limit = {
				OR = {
					has_title = title:e_wre

			NOT = { exists = title:e_byzantium.holder }
			NOT = { exists = title:h_eastern_roman_empire.holder }

			limit = {
				OR = {

			}
			NOT = { exists = title:e_wre.holder }
			NOT = { exists = title:h_western_roman_empire.holder }
		}
		trigger_else = { always = no }
		OR = {
			religion = religion:christianity_religion
			religion = religion:hellenism_religion
		}
		NOT = { has_claim_on = title:h_roman_empire }
		culture = { has_ability_to_reform_rome_directly_trigger = yes }
		NOT = { has_global_variable = sole_roman_empire }
	}
	
	is_valid = {
		OR = {
			has_trait = legendary_leader
			AND = {
				prestige_level >= 5
				legitimacy_level >= 5
			}
		}

		is_available_adult = yes
		is_at_war = no
	}

			if = {
				limit = { has_treasury = yes }
				add = 1000
			}
		}
		gold = {
			value = 0
			if = {
				limit = { has_treasury = no }
				add = 1000
			}
		}
		prestige = 1500
	}
	
	effect = {
		set_global_variable = { name = sole_roman_empire }
		show_as_tooltip = { add_pressed_claim = title:h_roman_empire }
		custom_tooltip = { text = roman_empire_claim_allows_for_conquest }

support_chariot_team_decision = {
	picture = {
		reference = "gfx/interface/illustrations/tournament_contest_illustrations/chariots_win_green.dds"
	}

	ai_check_interval_by_tier = {
		barony = 960
		county = 480
		duchy = 240
		kingdom = 180
		empire = 120
		hegemony = 60
	}
	
	desc = support_chariot_team_decision_desc
	selection_tooltip = support_chariot_team_decision_tooltip
	
	is_shown = {
		OR = {
			#Romans themselves will only do so if their culture supports it
			this.culture = { has_cultural_parameter = hosts_chariot_races }
			trigger_if = {
				limit = { is_landed = yes }
				#Foreigners, only if Roman Admin
				AND = {
					this.capital_county.culture = { has_cultural_parameter = hosts_chariot_races }
					OR = {
						government_has_flag = government_is_roman_dynastic
						government_has_flag = government_is_roman_administrative
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
		custom_description = {
			subject = root
			text = "has_changed_chariot_support_cooldown"
			NOT = { has_variable = changed_chariot_support }
		}
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "change_chariot_support_decision_confirm"
		show_from_start = yes

		item = {
			value = support_greens
			is_valid = { NOT = { has_trait = greens_supporter } }
			localization = support_greens_item_desc
			icon = "gfx/interface/icons/traits/greens_supporter.dds"
			ai_chance = { value = 50 }
		}

		item = {
			value = support_blues
			is_valid = { NOT = { has_trait = blues_supporter } }
			localization = support_blues_item_desc
			icon = "gfx/interface/icons/traits/blues_supporter.dds"
			ai_chance = { value = 50 }
		}

		item = {
			value = support_reds
			is_shown = { always = no }
			localization = support_reds_item_desc
			icon = "gfx/interface/icons/traits/reds_supporter.dds"
			ai_chance = { value = 0 }
		}

		item = {
			value = support_whites
			is_shown = { always = no }
			localization = support_whites_item_desc
			icon = "gfx/interface/icons/traits/whites_supporter.dds"
			ai_chance = { value = 0 }
		}
	}

	effect = {
		switch = {
			trigger = has_trait
			blues_supporter = { remove_trait = blues_supporter }
			greens_supporter = { remove_trait = greens_supporter }
			reds_supporter = { remove_trait = reds_supporter }
			whites_supporter = { remove_trait = whites_supporter }
			fallback = {}
		}
		if = {
			limit = { scope:support_greens = yes }
			custom_tooltip = {
				text = add_greens_supporter
				add_trait = greens_supporter
			}
		}
		else_if = {
			limit = { scope:support_blues = yes }
			custom_tooltip = {
				text = add_blues_supporter
				add_trait = blues_supporter
			}
		}
		else_if = {
			limit = { scope:support_reds = yes }
			custom_tooltip = {
				text = add_reds_supporter
				add_trait = reds_supporter
			}
		}
		else_if = {
			limit = { scope:support_whites = yes }
			custom_tooltip = {
				text = add_whites_supporter
				add_trait = whites_supporter
			}
		}
		hidden_effect = {
			set_variable = {
				name = changed_chariot_support
				months = @changed_chariot_support_cooldown_months
			}
		}
	}
	
	ai_will_do = {
		base = 0
		modifier = {
