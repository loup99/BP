##### Changed from Vanilla: Depends on graphical culture rather than geographical region
has_graphical_iberia_region_trigger = {
	trigger_if = {
		limit = {
			exists = capital_province
		}
		capital_province.culture = { has_building_gfx = iberian_building_gfx }
	}
	trigger_else_if = {
		limit = {
			NOT = { exists = capital_province }
		}
		trigger_if = { 
			limit = { exists = host.capital_province }
			host.capital_province.culture = { has_building_gfx = iberian_building_gfx }
		}
		trigger_else_if = {
			limit = {
				exists = liege.capital_province
			}
			liege.capital_province.culture = { has_building_gfx = iberian_building_gfx }
		}
		trigger_else_if = {
			limit = {
				exists = location.province_owner.capital_province
			}
			location.province_owner.capital_province.culture = { has_building_gfx = iberian_building_gfx }
		}
		trigger_else = { always = yes }
	}
	trigger_else = { always = yes }
}

##### Changed from Vanilla: Admin rulers 

byzantine_governor_trigger = {
	is_landed = yes
	is_independent_ruler = no # Fix vanilla bug
	government_allows = administrative
	liege ?= {
		OR = {
			is_roman_emperor_primary_title_trigger = yes
			primary_title = title:h_italy
			AND = {
				culture = {
					OR = {
						has_cultural_pillar = heritage_byzantine
						has_cultural_pillar = heritage_caucasian
						has_cultural_pillar = heritage_ancient_greek
					}
				}
				primary_title.tier >= tier_empire
			}
		}
	}
}

byzantine_governor_close_family_trigger = {
	location = prev.location
	byzantine_governor_trigger = yes
}

east_asian_governor_trigger = {
	is_landed = yes
	is_governor = yes
	government_allows = administrative
	top_liege ?= {
		OR = {
			primary_title = title:h_china
			primary_title.de_jure_liege = title:h_china
			AND = {
				culture_has_east_asian_heritage_pillar_trigger = yes
				primary_title.tier >= tier_empire
			}
		}
	}
}

east_asian_governor_close_family_trigger = {
	location = prev.location
	east_asian_governor_trigger = yes
}
