#General Triggers
major_war_type_trigger = {
	$WAR$ = {
		OR = {
			using_cb = bp_migration_war_cb
		}
	}
}
#Varangian Conquest Trigger: Only available after migrations are over
bp_varangrian_conquest_trigger = { current_date >= 800.1.1 }
cb_prestige_level_trigger = {
	trigger_if = {
		limit = { culture = { has_cultural_parameter = cb_lower_prestige_level_requirement } }
		prestige_level >= $EASIER_PRESTIGE_LEVEL$
	}
	trigger_else = { prestige_level >= $BASE_PRESTIGE_LEVEL$ }
}
#Because we need to make the federates follow very specific constraints when it comes to wars
bp_federate_war_validity_trigger = {
	trigger_if = {
		limit = {
			government_has_flag = government_is_roman_federate
			exists = scope:defender
		}
		OR = {
			#Case 1: Not of the same realm
			scope:defender.top_liege != scope:attacker.top_liege
			#Case 2: Same Realm, but is not administrative
			AND = {
				scope:defender.top_liege = scope:attacker.top_liege
				NOT = { scope:defender = { government_allows = administrative } }
			}
			#Case 3: Same Realm, administrative but is a federate
			AND = {
				scope:defender.top_liege = scope:attacker.top_liege
				scope:defender = { government_allows = administrative }
				scope:defender = { government_has_flag = government_is_quasi_feudal }
			}
		}
	}
}

bp_migration_war_distace_tier_logic_trigger = {
	OR = {
		#100k is about the distance from Constantinople to Chios or nearly to the Danube
		# We base everything on this distance, keeping mind the actual distance is the square-root
		# Of this value (i.e., 10k = 100)
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 4 }
			scope:target = {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					holder = { in_diplomatic_range = root }
				}
			}
		}
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 3 }
			scope:target = {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					squared_distance = { target = root.capital_county value < 640000 } #800
				}
			}
		}
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 2 }
			scope:target = {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					squared_distance = { target = root.capital_county value < 160000 } #400
				}
			}
		}
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 1 }
			scope:target = {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					squared_distance = { target = root.capital_county value < 40000 } #200
				}
			}
		}
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 0 }
			scope:target = {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					squared_distance = { target = root.capital_county value < 10000 } #100
				}
			}
		}
		#Neighboring Kingdom is always valid
		scope:target = {
			any_in_de_jure_hierarchy = {
				tier = tier_county
				any_title_to_title_neighboring_and_across_water_county = {
					holder = { target_is_same_character_or_above = root }
				}
			}
		}
	}
}
