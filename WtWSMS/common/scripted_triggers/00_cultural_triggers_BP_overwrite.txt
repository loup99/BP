# Checks if you could Hybridise with the culture without checking for acceptance
is_valid_for_hybridising_trigger = {
	$CHARACTER$ = {
		##### Changed from Vanilla vvvvv
		trigger_if = {
			limit = {
				general_situation_regional_parameter_trigger = {
					CHARACTER = $CHARACTER$
					PARAMETER = faster_divergence_hybridization_time
				}
			}
			culture = {
				NOT = { has_same_culture_heritage = $CULTURE$ }
				culture_age >= BP_culture_hybrid_cooldown_halved
			}
		}
		trigger_else = {
			culture = {
				NOT = { has_same_culture_heritage = $CULTURE$ }
				culture_age >= culture_hybrid_cooldown
			}
		}
		##### Changed from Vanilla ^^^^^
		trigger_if = {
			limit = {
				NAND = {
					exists = var:special_culture
					var:special_culture = $CULTURE$
				}
			}
			custom_description = {
				text = culture_present_in_realm_for_hybridization
				any_sub_realm_county = {
					culture = $CULTURE$
				}
			}
		}
		custom_description = {
			text = culture_is_not_parent_culture
			culture = {
				NOT = {
					any_parent_culture = {
						this = $CULTURE$
					}
				}
			}
		}
		custom_description = {
			text = culture_is_not_child_culture
			$CULTURE$ = {
				NOT = {
					any_parent_culture = {
						this = $CHARACTER$.culture
					}
				}
			}
		}
		##### Changed from Vanilla vvvvv
		trigger_if = { #Shared Kulturbund blocks hybridization except for Malleable Invaders/Culture Blending by the actor
			limit = { culture = { has_same_kulturbund_as_trigger = { TARGET = $CULTURE$ } } }
			custom_description = {
				text = must_have_easier_to_hybrdize_parameter_to_hybridize_kulturbund
				culture = { has_cultural_parameter = enables_kulturbund_hybridization }
			}
		}
		custom_description = {
			text = cannot_combine_hybridization_blocking_tradition_cultures
			NAND = {
				culture = { has_cultural_parameter = greco_roman_social_tradition }
				$CULTURE$ = { has_cultural_parameter = greco_roman_social_tradition }
			}
		}
		custom_description = {
			text = cannot_combine_hybridization_blocking_tradition_cultures
			NAND = {
				culture = { has_cultural_parameter = greco_roman_social_tradition }
				$CULTURE$ = { has_cultural_parameter = greco_roman_social_tradition }
			}
		}
		trigger_if = {
			limit = { culture = { has_cultural_parameter = monolithic_culture_feature } }
			custom_description = {
				text = must_have_duchy_or_better_to_hybridize
				primary_title.tier >= tier_duchy
			}
		}
		custom_description = {
			text = cannot_hybridize_culture_drift
			NOR = {
				#Can't combine Latin + Romance
				AND = {
					culture = { is_romance_heritage_trigger = yes }
					$CULTURE$ = { has_cultural_pillar = heritage_latin }
				}
				AND = {
					$CULTURE$ = { is_romance_heritage_trigger = yes }
					culture = { has_cultural_pillar = heritage_latin }
				}
				#Can't combine Migration + Medieval Germanic
				AND = {
					culture = { is_migratory_germanic_heritage_trigger = yes }
					$CULTURE$ = { is_medieval_germanic_heritage_trigger = yes }
				}
				AND = {
					$CULTURE$ = { is_migratory_germanic_heritage_trigger = yes }
					culture = { is_medieval_germanic_heritage_trigger = yes }
				}
			}
		}
		##### Changed from Vanilla ^^^^^
	}
}

##### Changed from Vanilla: Remove ability to change language based on legends
language_is_shown_trigger = {
	scope:character = {
		OR = {
			AND = {
				exists = var:legendary_culture
				var:legendary_culture = { has_cultural_pillar = $LANGUAGE$ }
			}
			AND = {
				# The target language needs to fulfill ALL of the following:
				any_sub_realm_county = { # Language needs to be well-represented within your realm
					percent >= 0.1
					culture = { has_cultural_pillar = $LANGUAGE$ }
				}
				any_sub_realm_county = { # Language needs to border your own culture
					culture = { has_cultural_pillar = $LANGUAGE$ }
					any_neighboring_county = {
						culture = { this = scope:character.culture }
					}
				}
				any_sub_realm_county = { # You need a minimum acceptance with any culture speaking the language
					culture = {
						has_cultural_pillar = $LANGUAGE$
						cultural_acceptance = {
							target = scope:character.culture
							value >= 25
						}
					}
				}
			}
		}
	}
}
##### Changed from Vanilla: Remove ability to change heritage based on legends
heritage_is_shown_trigger = {
	scope:character = {
		OR = {
			AND = {
				exists = var:legendary_culture
				var:legendary_culture = { has_cultural_pillar = $HERITAGE$ }
			}
			culture = { has_cultural_pillar = $HERITAGE$ }
		}
	}
}
