valid_demand_conversion_conditions_trigger = {
	scope:recipient = {
		NOR = {
			has_strong_hook = scope:puppet_or_actor
			has_trait = devoted
			has_trait = order_member
			is_imprisoned_by = scope:puppet_or_actor
			is_at_war_with = scope:puppet_or_actor
		}
	}
	scope:puppet_or_actor = {
		trigger_if = {
			limit = {
				government_has_flag = government_is_landless_adventurer
				scope:recipient = { is_courtier_of = scope:puppet_or_actor }
			}
			custom_tooltip = {
				text = valid_demand_conversion_conditions_trigger.tt.domicile_requires_shrine
				domicile = { has_domicile_parameter = camp_enable_conversion }
			}
		}
		custom_tooltip = {
			text = demand_conversion_nomadic_zealous_tt
			NAND = {
				has_trait = nomadic_philosophy
				has_trait = zealous
			}
		}
	}

	custom_description = {
		text = "is_head_of_religion"
		subject = scope:recipient
		scope:recipient.faith.religious_head != scope:recipient
	}
	custom_description = {
		text = "is_protected_via_contract"
		subject = scope:recipient
		# Vassal Contract forbids meddling by liege
		NAND = {
			exists = scope:recipient.liege
			scope:recipient.liege = scope:puppet_or_actor
			scope:recipient = {
				is_ruler = yes
				vassal_contract_has_flag = religiously_protected
			}
		}
	}
	trigger_if = {
		limit = {
			scope:recipient = {
				AND = {
					has_variable = cannot_be_converted_by_value
					var:cannot_be_converted_by_value = scope:puppet_or_actor
				}
			}
		}
		custom_tooltip = {
			text = promised_to_not_convert_character
			scope:recipient = {
				NOR = {
					has_variable = cannot_be_converted_by_value
					var:cannot_be_converted_by_value = scope:puppet_or_actor
				}
			}
		}
	}
	custom_tooltip = {
		text = cannot_take_overt_hostile_actions_against_diarch.tt
		NOT = { scope:recipient ?= scope:puppet_or_actor.diarch }
	}
	##### Changed from Vanilla vvvvv
	BP_demand_conversion_trigger = yes
	##### Changed from Vanilla ^^^^^
}

##### Changed from Vanilla: Tied into WtWSMS Mechanics
can_create_rite_trigger = {

	is_adult = yes
	is_at_war = no
	
	trigger_if = {
		limit = {
			is_clergy = no
			highest_held_title_tier = tier_county
			top_liege != this
		}
		custom_description = {
			text = "faith_creation_duchy_or_higher"
			highest_held_title_tier >= tier_duchy
		}
	}

	trigger_if = {
		limit = {
			rite = {
				rite_has_doctrine = doctrine_theocracy_temporal
			}
			is_clergy = no
			OR = {
				has_puppet_type = theological_agent_puppet
				has_puppet_type = external_theological_agent_puppet
			}
			# Allow Mandala God-Kings and Son of Heaven to create and edit rites freely
			trigger_if = {
				limit = {
					top_overlord = this
				}
				NOR = {
					government_has_flag = government_is_mandala
					government_has_flag = government_is_celestial
				}
			}
		}
		is_puppet_action_valid = puppet_trigger_can_create_rite
	}
	
	custom_description = {
		text = "character_is_not_real_head"
		faith.religious_head != root
	}

	custom_description = {
		text = "character_can_only_create_one_rite"
		NOT = { exists = var:has_created_a_faith }
	}

	trigger_if = {
		limit = {
			faith = { has_doctrine_parameter = unreformed }
		}

		NOT = {
			custom_description = {
				text = faith_has_been_reformed
				object = faith
				exists = faith.var:has_been_reformed
			}
		}

		trigger_if = {
			limit = {
				NOT = {
					has_character_flag = ignore_3_holy_site_rule
				}
			}
			# Empire Faith Gate: easier reformation path
			trigger_if = {
				limit = {
					empire_faith_gate_enabled_trigger = yes
				}
				custom_tooltip = {
					text = empire_faith_gate_reformation_piety_level
					piety_level >= 2
				}
				custom_tooltip = {
					text = empire_faith_gate_reformation_tribal_authority
					has_realm_law = tribal_authority_3
				}
				custom_tooltip = {
					text = empire_faith_gate_reformation_holy_site
					faith = {
						any_holy_site_barony = {
							count >= 3
							county.holder = {
								faith = root.faith
							}
						}
					}
				}
			}
			# Vanilla: 3 holy sites
			trigger_else = {
				faith = {
					num_realm_holy_sites_faithful_holders = {
						CHARACTER = root
						COUNT = 3
					}
				}
			}
		}

		custom_tooltip = {
			text = faith_reformation_via_decision
			faith = {
				NOT = { has_variable = block_reformation_var }
			}
		}
		##### Changed from Vanilla vvvvv
		trigger_if = {
			limit = { rite = { rite_has_parameter = unreformed } }
			culture = { has_innovation = innovation_formalized_rituals }
		}
		trigger_else = { culture = { has_innovation = innovation_state_religion } }
		##### Changed from Vanilla ^^^^^
	}
}

faith_is_syncretic_with = {
	$FAITH$ = {
		trigger_if = {
			limit = { has_doctrine_parameter = unreformed }
			prev = { has_doctrine_parameter = unreformed_syncretic_actor }
		}
		trigger_else_if = {
			limit = { religion_tag = christianity_religion }
			prev = { has_doctrine_parameter = christian_syncretic_actor }
		}
		trigger_else_if = {
			limit = { religion_tag = islam_religion }
			prev = { has_doctrine_parameter = islamic_syncretic_actor }
		}
		trigger_else_if = {
			limit = { religion_tag = judaism_religion }
			prev = { has_doctrine_parameter = jewish_syncretic_actor }
		}
		trigger_else_if = {
			limit = { is_in_family = rf_eastern }
			prev = { has_doctrine_parameter = eastern_syncretic_actor }
		}
		trigger_else_if = {
			limit = { is_in_family = rf_sinitic }
			prev = { has_doctrine_parameter = sinitic_syncretic_actor }
		}
		##### Changed from Vanilla vvvvv
		trigger_else_if = {
			limit = { religion_tag = hellenism_religion }
			prev = { has_doctrine_parameter = hellenism_syncretic_actor }
		}
		trigger_else_if = {
			limit = { has_doctrine = special_doctrine_is_egyptian_faith }
			prev = { has_doctrine_parameter = egyptian_syncretic_actor }
		}
		trigger_else_if = {
			limit = { has_doctrine = special_doctrine_is_zoroastrian_faith }
			prev = { has_doctrine_parameter = zoroastrian_syncretic_actor }
		}
		##### Changed from Vanilla ^^^^^
		trigger_else = {
			always = no
		}
	}
}
