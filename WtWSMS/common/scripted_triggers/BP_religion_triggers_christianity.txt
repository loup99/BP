general_liturgical_rite_pick_trigger = {
	trigger_if = {
		limit = { scope:actor = { is_ai = yes } }
		doctrine:$LITURGICAL_RITE$ = { is_in_list = selected_doctrines }
	}
	trigger_else = {
		OR = {
			#Your rite uses Latin, the main rite does, or you speak a Romance language
			scope:actor.rite = { rite_has_parameter = $SIMILAR_RITE_TRIGGER$ }
			scope:actor.rite.faith.main_rite = { rite_has_parameter = $SIMILAR_RITE_TRIGGER$ }
			scope:actor.culture = { has_cultural_parameter = $LANGUAGE_GROUP$ }
			#It is already selected (aka, at start)
			doctrine:$LITURGICAL_RITE$ = { is_in_list = selected_doctrines }
		}
		custom_description = {
			text = incompatible_tenet_ecstatic_prophecy_trigger
			NOT = { tenet:tenet_ecstatic_prophecy = { is_in_list = selected_doctrines } }
		}
	}
}

latin_liturgical_rite_pick_trigger = {
	general_liturgical_rite_pick_trigger = {
		LITURGICAL_RITE = $LITURGICAL_RITE$
		SIMILAR_RITE_TRIGGER = latin_rites_are_similar
		LANGUAGE_GROUP = language_group_romance
	}
}
