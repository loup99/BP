tradition_greek_succession = { #Byzantine Succession Variant for non-Roman Reformation cultures
	category = regional

	layers = {
		0 = intrigue
		1 = mediterranean
		4 = crown.dds
	}
	
	is_shown = {
		AND = {
			has_cultural_pillar = heritage_byzantine
			scope:character = {
				NOR = {
					has_title = title:e_byzantium
					has_title = title:h_roman_empire
					has_title = title:h_eastern_roman_empire
				}
			}
		}
	}
	can_pick = {
		traditions_legal_tradition_is_in_list_trigger = { TRADITION = culture_tradition:tradition_greek_succession }
		traditions_body_status_type_is_in_list_trigger = { TRADITION = culture_tradition:tradition_greek_succession }
	}
	can_pick_for_hybridization = {
		traditions_legal_tradition_is_in_list_trigger = { TRADITION = culture_tradition:tradition_greek_succession }
		traditions_body_status_type_is_in_list_trigger = { TRADITION = culture_tradition:tradition_greek_succession }
		custom_description = {
			text = hybrid_must_have_byzantine_heritage_for_tradition_greek_succession
			culture_pillar:heritage_byzantine = { is_in_list = traits }
		}
	}
	
	parameters = {
		is_a_legal_tradition = yes
		can_blind_prisoners = yes
		can_castrate_prisoners = yes
		eunuch_trait_bonuses = yes
		can_appoint_chief_eunuch = yes
		physical_disfigurement_blocks_inheritance = yes #Actually only blocks claimants and reduce score in elective. Also includes infertility for tooltip reasons.
		disloyal_trait_more_common = yes
	}
	
	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_courtly = { is_in_list = traits }
						culture_pillar:ethos_communal = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_courtly_or_communal_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 100
		if = { #Not Roman government, so *ick*
			limit = {
				NOR = {
					scope:character = { has_main_roman_government_trigger = yes }
					scope:character = { has_subroman_government_trigger = yes }
				}
			}
			multiply = 0.0
		}
	}
}

tradition_roman_politics = {
	category = regional

	layers = {
		0 = diplo
		1 = mediterranean
		4 = palace_politics.dds 
	}

	is_shown = {
		has_ep3_dlc_trigger = yes
		OR = {
			has_cultural_pillar = heritage_latin
			scope:character = { has_title = title:h_western_roman_empire }
			scope:character = { has_title = title:h_roman_empire }
		}
		NOT = { has_cultural_tradition = tradition_roman_succession }
		NOT = { has_cultural_tradition = tradition_byzantine_succession }
		NOT = { has_cultural_tradition = tradition_byzantine_politics }
	}
	can_pick = {
		custom_tooltip = {
			text = cannot_have_tradition_court_eunuchs
			NOT = { culture_tradition:tradition_court_eunuchs = { is_in_list = traits } }
		}
		custom_tooltip = {
			text = cannot_have_tradition_legalistic
			NOT = { culture_tradition:tradition_legalistic = { is_in_list = traits } }
		}
	}
	can_pick_for_hybridization = {
		custom_tooltip = {
			text = cannot_have_tradition_court_eunuchs
			NOT = { culture_tradition:tradition_court_eunuchs = { is_in_list = traits } }
		}
		custom_tooltip = {
			text = cannot_have_tradition_legalistic
			NOT = { culture_tradition:tradition_legalistic = { is_in_list = traits } }
		}
	}
	
	#NB: Latin Romans _used_ Eunuchs, but were very squeamish about actually castrating
	# so they get the bonuses but can't actually make them themselves
	parameters = {
		is_a_legal_tradition = yes
		
		cheaper_to_change_laws = yes
		
		reduced_bodyguard_aptitude = yes
		eunuch_trait_bonuses = yes
		
		ambitious_trait_more_common = yes
		disloyal_trait_more_common = yes
		
		cultrad_unlocks_t2_countermeasures = yes
		
		holds_triumphs = yes
		
		can_restore_roman_empire = yes
		unlock_innovation_roman_civil = yes
		roman_legal_traits_are_valued = yes
		enables_senate_dynasty_elevation = yes
	}
	
	character_modifier = {
		monthly_influence_mult = 0.1
		vassal_limit = 30
		owned_hostile_scheme_success_chance_add = 5
		political_scheme_phase_duration_add = minor_scheme_phase_duration_bonus_value
		tyranny_gain_mult = 0.25
		monthly_prestige_gain_per_dread_mult = -0.01
	}
	
	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_bureaucratic = { is_in_list = traits }
						culture_pillar:ethos_courtly = { is_in_list = traits }
						culture_pillar:ethos_spiritual = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bureaucratic_courtly_or_spiritual_desc
				}
			}
			if = {
				limit = {
					has_dlc_feature = royal_court
					OR = {
						scope:character = { has_royal_court = no }
						NOT = {
							scope:character = {
								court_grandeur_current_level >= very_high_court_grandeur_level
							}
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = culture_head_court_grandeur_desc
				}
			}

			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_roman_politics }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}
	
	ai_will_do = {
		value = 200
		if = {
			limit = { # Admin should be more into this
				scope:character = {
					government_allows = administrative
				}
			}
			multiply = 2
		}
	}
}
