##### Variant of tradition_devoted_horsemanship, but for Western Steppe only
tradition_devoted_horsemanship_western_steppe = {
	category = regional

	layers = {
		0 = martial
		1 = steppe
		4 = audacious_cadets.dds
	}

	is_shown = {
		scope:character.culture = { culture_overlaps_geographical_region = world_steppe_west }
		NOT = { has_cultural_tradition = tradition_devoted_horsemanship }
		NOT = { has_cultural_tradition = tradition_light_cavalry_western_steppe }
	}

	can_pick = {
		custom_description = {
			text = half_of_counties_in_world_steppe_west
			any_culture_county = {
				percent > 0.50
				any_province = { geographical_region = world_steppe_west }
			}
		}
		OR = {
			scope:character = { mpo_can_recruit_nomad_maa_trigger = yes }
			scope:character = { has_trait = nomadic_philosophy }
		}
	}

	parameters = {
		unlock_maa_horse_archers = yes
		unlock_maa_sarmatian_lancers = yes
		can_recruit_open_specialist = yes
		unlock_horse_pastures_building = yes
		unlock_voluntary_laampdom = yes
		stewardship_traits_give_out_cav_bonus = yes
	}

	character_modifier = {
		movement_speed = 0.1
		character_travel_speed_mult = 0.2
		movement_speed_land_raiding = 0.2
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				multiply = 2
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_bellicose = { is_in_list = traits }
						culture_pillar:ethos_communal = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_or_communal_desc
				}
			}

			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_devoted_horsemanship_western_steppe }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}

			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 400
	}
}

#Captures the Cranial deformation. Functions as Ruling Caste-like, isolationist tradition
tradition_cranial_deformation = {
	category = regional

	layers = {
		0 = steward
		1 = steppe
		4 = greeting.dds
	}
	
	is_shown = { has_cultural_tradition = tradition_cranial_deformation }
	can_pick = {
		#government_has_flag = government_is_nomadic
		OR = {
			has_cultural_tradition = tradition_cranial_deformation
			culture_tradition:tradition_cranial_deformation = { is_in_list = traits }
		}
		custom_tooltip = {
			text = cannot_have_tradition_ruling_caste
			NOT = { culture_tradition:tradition_ruling_caste = { is_in_list = traits } }
		}
	}
	can_pick_for_hybridization = {
		#government_has_flag = government_is_nomadic
		OR = {
			has_cultural_tradition = tradition_cranial_deformation
			culture_tradition:tradition_cranial_deformation = { is_in_list = traits }
		}
		custom_tooltip = {
			text = cannot_have_tradition_ruling_caste
			NOT = { culture_tradition:tradition_ruling_caste = { is_in_list = traits } }
		}
	}
	
	parameters = {
		courtiers_less_likely_to_leave_same_culture_court = yes
		peasant_and_populist_factions_less_common = yes
		less_likely_to_culture_convert = yes
		cranial_deformation_give_bonus = yes
	}
	culture_modifier = {
		cultural_acceptance_gain_mult = -0.30
	}
	character_modifier = {
		same_culture_opinion = 5
	}
	
	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_spiritual = { is_in_list = traits }
						culture_pillar:ethos_courtly = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_spiritual_or_courtly_desc
				}
			}
			if = {
				limit = {
					scope:character = {
						any_councillor = {
							NOR = {
								culture = scope:character.culture
								is_spouse_of = scope:character # Might be hard to have the right culture for the Spouse; but we can remove this helper if needed
							}
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = has_other_cultures_in_council_desc
				}
			}
			if = {
				limit = { has_variable = { name = aggressive_traditionalist_discount } }
				add = {
					value = traditionalist_discount
					desc = traditionalist_discount_desc
				}
			}
			
			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_cranial_deformation }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}
	
	ai_will_do = { 
		value = 100
		if = {
			limit = {
				scope:character.dread < 25
			}
			multiply = 0.5
		}
		if = {
			limit = {
				scope:character.tyranny < 25
			}
			multiply = 0.5
		}
	}
}
