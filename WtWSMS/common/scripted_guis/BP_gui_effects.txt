reset_custom_map_mode_effect = {
	effect = {
		if = {
			limit = { exists = global_var:custom_map_mode }
			remove_global_variable = custom_map_mode
		}
		every_county = {
			limit = { is_landless_type_title = no }
			save_temporary_scope_as = the_county
			every_county_province = { barony = { set_color_from_title = scope:the_county } }
		}
	}
}
