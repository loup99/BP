	}
	
	traits = {}
	
	ai_chance = {
		value = 11

			limit = { can_start_new_legacy_track_trigger = no }

					government_has_flag = is_a_roman_government
					dynasty = dynasty:1042165 #Amaling

					dynasty = dynasty:1042160 #Thorcilings
					dynasty = dynasty:1042146 #Burgundians
					#Karlings
					dynasty = dynasty:25061
					dynasty = dynasty:25065
					dynasty = dynasty:25066
					#Your ancestors were once civilized...
					dynasty.dynasty_founder.culture = { has_cultural_parameter = greco_roman_social_tradition }
					dynasty.dynasty_founder.culture = { has_cultural_parameter = can_restore_roman_empire }
					dynasty.dynasty_founder.culture = { has_cultural_parameter = baseline_chinese_acceptance_increased }
					dynasty.dynasty_founder.culture = { has_cultural_tradition = tradition_tgp_inward_perfection }
				}
			}
			multiply = 0
		}
		#AI will tend to do this for certain cultures/dynasties
		if = {
			limit = {
				OR = {
					culture = culture:vandalic
					dynasty.dynasty_founder.culture = culture:vandalic
					dynasty = dynasty:1042147 #Baltings
				}
			}
			multiply = 9999
		}
	}
