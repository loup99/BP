#10% every 5 years implies a bit more than 50% chance every 35 years (7x5 year blocks)
#7% every 5 years implies a bit more than 50% chance every 50 years (10x5 year blocks)
#Because of how logic is set in random_events, we fire the no_event stuff if the trigger isn't there
@admin_to_feudal_base_chance = 10
@admin_to_feudal_change_reduction = 30


five_year_playable_pulse = {
	on_actions = {
		delay = { days = { 30 1095 } } #one month to 3 years
		BP_admin_to_feudal_mutation
		BP_admin_to_feudal_aristocracy_solidification
		BP_admin_to_feudal_military_landing
		BP_admin_to_feudal_domestic_taxation
	}
}

BP_admin_to_feudal_mutation = {
	trigger = { bp_meets_requirements_to_fedualize_trigger = yes }
	random_events = {
		chance_to_happen = @admin_to_feudal_base_chance
		chance_of_no_event = {
			value = 0
			if = {
				limit = {
					general_situation_regional_parameter_trigger = {
						CHARACTER = this
						PARAMETER = increased_subroman_decay
					}
				}
				add = @admin_to_feudal_change_reduction
			}
		}
		100 = BP_admin_to_feudal_events.0050
	}
}
