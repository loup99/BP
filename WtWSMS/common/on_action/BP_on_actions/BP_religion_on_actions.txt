# Code on-action: character creates a faith
# Root is the creator
# scope:old_faith is the faith they used to have
on_faith_created = {
	on_actions = { BP_inherit_ethnic_religions }
}

BP_inherit_ethnic_religions = {
	effect = {
		if = {
			limit = { scope:old_faith = { has_variable = ethnic_religion_culture } }
			root.faith = { set_variable = { name = ethnic_religion_culture value = scope:old_faith.var:ethnic_religion_culture } }
		}
	}
}

# Code on-action: character converts faith. Only fired when using the convert window, or the script effect that uses the same logic
# Root is the converter
# scope:old_faith is the faith they used to have
on_faith_conversion = {
	events = {
		bp_faith_conversion.0001 #Culture keeps its old marriage customs when going from unreformed to reformed faith
	}
	on_actions = {
		BP_check_zealous_pagans
		BP_change_faith_dynasty_coa
		BP_adult_credobaptism
	}
}

BP_check_zealous_pagans = {
	effect = {
		if = {
			limit = {
				OR = {
					root.culture = { has_cultural_tradition = tradition_zealous_pagans }
					root.culture = { has_cultural_tradition = tradition_zealous_pagans_non_dlc }
				}
			}
			#If it's a zealous pagan converting, got to check if it gets removed
			root.culture = { trigger_event = { id = bp_change_tradition_events.0003 days = { 1 365 } } }
		}
	}
}

BP_change_faith_dynasty_coa = {
	effect = {
		if = {
			limit = {
				NOT = { root.dynasty = { has_variable = BP_changed_faith_dynasty_coa } }
				root.dynasty = dynasty:1042151
				root.dynasty.dynast = root
				root.faith.religion = religion:christianity_religion
			}
			dynasty = {
				set_coa = title:k_france
				set_variable = BP_changed_faith_dynasty_coa
			}
			house = { set_coa = dynasty:k_france }
		}
	}
}

BP_adult_credobaptism = {
	trigger = {
		rite = { rite_has_parameter = adult_conversion_bonus }
		trigger_if = {
			limit = { has_variable = credobaptism_faith }
			NOT = { this.var:credobaptism_faith = this.rite.faith }
		}
	}
	effect = {
		set_variable = {
			name = credobaptism_faith
			value = scope:the_char.rite.faith
		}
		send_interface_toast = {
			type = event_toast_effect_good
			title = tenet_credobaptism_toast
			left_icon = scope:the_char
			add_piety = major_piety_value
			set_variable = {
				name = credobaptism_faith
				value = scope:the_char.rite.faith
			}
		}
	}
}

# Code on-action: when a character selects a new personal tenet successfully this onaction is called.
# Root is character picking the tenet
# scope:tenet is the tenet they just picked
on_personal_tenet_gain = {
	on_actions = { BP_personal_tenet_gain_effects }
}

BP_personal_tenet_gain_effects = {
	effect = {
		if = {
			limit = {
				scope:tenet = tenet:tenet_total_depravity
				has_personal_tenet_flag = gain_level_of_piety_when_adopting_this_tenet
			}
			add_piety_level = 1
		}
	}
}

on_birthday = {
	on_actions = {
		bp_religious_birthday_actions
	}
}

bp_religious_birthday_actions = {
	first_valid_on_action = {
		bp_on_16th_birthday
	}
}

bp_on_16th_birthday = {
	trigger = { is_adult = yes }
	effect = {
		save_temporary_scope_as = the_char
		if = {
			limit = { rite = { rite_has_parameter = adult_baptism_bonus } }
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_good
					title = tenet_credobaptism_toast
					left_icon = scope:the_char
					add_piety = massive_piety_value
					set_variable = {
						name = credobaptism_faith
						value = scope:the_char.rite.faith
					}
				}
			}
		}
	}
}
