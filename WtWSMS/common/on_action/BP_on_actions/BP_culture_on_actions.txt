# Happens when a culture is created through hybridization or divergence
# Root = The new culture
# scope:founder = The founder of the new culture
on_culture_created = {
	on_actions = {
		bp_setup_necessary_divergence_name_tracking
		bp_apply_special_hybrid_cases
		bp_remove_certain_innovations
		bp_inherit_certain_variables
	}
}

bp_setup_necessary_divergence_name_tracking = {
	#Only divergent cultures have roots; hybrids are their own roots when diverging away from
	trigger = { is_divergent_culture = yes }
	effect = {
		#If a root already exists, track it.
		if = {
			limit = { any_parent_culture = { has_variable = root_culture } }
			random_parent_culture = {
				limit = { has_variable = root_culture }
				save_scope_as = the_parent_culture
			}
			#Track the root culture
			set_variable = { name = root_culture value = scope:the_parent_culture.var:root_culture }
		}
		#Otherwise, our parent is a root culture
		else = {
			random_parent_culture = { save_scope_as = the_parent_culture }
			set_variable = { name = root_culture value = scope:the_parent_culture }
		}
	}
}

bp_apply_special_hybrid_cases = {
	effect = {
		#Glorious East Celtic Namelist!
		random_parent_culture = {
			limit = { has_same_culture_heritage = root }
			save_scope_as = parent_1
		}
		random_parent_culture = {
			limit = {
				is_hybrid_culture = yes
				NOT = { this = scope:parent_1 }
			}
			save_scope_as = parent_2
		}
		if = {
			limit = {
				is_hybrid_culture = yes
				exists = scope:parent_2
				hybrid_has_new_namelist_trigger = yes
			}
			scope:founder = { trigger_event = bp_culture_emergence_events.1001 }
		}
	}
}

bp_remove_certain_innovations = {
	effect = {
		if = {
			limit = {
				NOT = { has_cultural_parameter = unlock_innovation_roman_civil }
				has_innovation = innovation_roman_administration
			}
			remove_innovation = innovation_roman_administration
		}
		if = {
			limit = {
				NOT = { has_cultural_parameter = unlock_innovation_roman_civil }
				has_innovation = innovation_roman_concrete
			}
			remove_innovation = innovation_roman_concrete
		}
		if = {
			limit = {
				NOT = { has_cultural_parameter = unlock_innovation_roman_civil }
				has_innovation = innovation_legal_codex
			}
			remove_innovation = innovation_legal_codex
		}
	}
}

bp_inherit_certain_variables = {
	trigger = {
		is_hybrid_culture = yes
		any_parent_culture = {
			has_variable = passed_cultural_law
			has_same_culture_heritage = prev
		}
	}
	effect = {
		random_parent_culture = {
			limit = {
				has_variable = passed_cultural_law
				has_same_culture_heritage = prev
			}
			save_temporary_scope_as = law_culture
		}
		set_variable = { name = passed_cultural_law value = scope:law_culture }
	}
}

# Happens when a culture gains a tradition
# Root = culture
# scope:tradition = flag for the tradition added
on_tradition_added = {
	on_actions = { bp_update_same_religion_succession_law }
}

bp_update_same_religion_succession_law = {
	effect = {
		every_ruler = {
			limit = {
				is_ruler = yes
				should_have_same_religion_succession_trigger = yes
				NOT = { has_realm_law = same_religion_only_law }
			}
			add_realm_law_skip_effects = same_religion_only_law
		}
	}
}

# Happens once a year for the culture. Arbitrary day, but the same day every year for that culture
# Root is the culture
yearly_culture_pulse = {
	trigger = {
		is_a_living_culture_trigger = yes
		OR = {
			has_cultural_tradition = tradition_elite_rulership
			has_cultural_tradition = tradition_tribal_hierarchy
			has_cultural_tradition = tradition_zealous_pagans
			has_cultural_tradition = tradition_zealous_pagans_non_dlc
			has_cultural_tradition = tradition_early_slavic_traditions
		}
	}
	#This event fires because `on_culture_era_changed` gets referenced in history, which leads to
	#scoping errors. So we have a yearly event firing instead...
	events = {
		delay = { days = { 1 365 } }
		bp_slavic_traditions_change.0001
		bp_change_tradition_events.0001 #Remove Migratory Culture Tradition
		bp_change_tradition_events.0002 #Tribal Hierarchy to Hereditary Hierarchy Tradition
		bp_change_tradition_events.0003 #Zealous Pagans Tradition Removal
		
	}
}

historical_culture_ashkenazi = {
	trigger = { NOT = { has_global_variable = ashkenazi_culture_already_exists } }
	effect = {
		save_temporary_scope_value_as = { name = drift_start_year value = 750 }
		save_temporary_scope_value_as = { name = drift_period_years value = 150 }
		random = {
			chance = 0
			# Melting pot logic
			modifier = { add = 0.25 current_year >= drift_start_year_0_period }
			modifier = { add = 0.25 current_year >= drift_start_year_1_period }
			modifier = { add = 0.50 current_year >= drift_start_year_2_period }
			modifier = { add = 1.00 current_year >= drift_start_year_3_period }
			modifier = { add = 2.00 current_year >= drift_start_year_4_period }
			modifier = { add = 2.00 current_year >= drift_start_year_5_period }
			modifier = { add = 3.00 current_year >= drift_start_year_6_period }
			modifier = { add = 5.00 current_year >= drift_start_year_7_period }
			# Accelerate timeline if Crusades have been called for
			modifier = { add = 5.00 religion:christianity_religion = { exists = var:variable_ghw_unlocked } }
			
			# Check for a non-Romance culture in Europe in the early middle ages
			random_culture_global = {
				limit = {
					exists = this.culture_head
					# Early medieval era
					has_cultural_era_or_later = culture_era_early_medieval
					# Needs to be culturally distinct - not Roman or Jewish
					NOR = {
						has_cultural_pillar = heritage_byzantine
						has_cultural_pillar = language_greek
						has_cultural_pillar = heritage_latin
						has_cultural_parameter = language_group_romance
						has_cultural_pillar = heritage_israelite
					}
					# Must be in central Europe
					any_culture_county = {
						percent >= 0.5
						any_province = { geographical_region = custom_ashkenaz }
					}
				}
				weight = {
					base = 1
					modifier = { add = this.culture_number_of_counties }
				}
				save_scope_as = patron_culture
				set_global_variable = ashkenazi_culture_already_exists
				
				# Set up culture
				change_cultural_acceptance = {
					target = culture:ashkenazi
					value = 100
					desc = ashkenazi_culture_created
				}
				culture:ashkenazi = {
					reset_culture_creation_date = yes
					get_all_innovations_from = culture:tzarfati
					get_all_innovations_from = scope:patron_culture
					set_language_from = scope:patron_culture
				}
				
				# Add minorities and exiles (inviting Ashkenazim occurred historically)
				scope:patron_culture.culture_head.capital_county = {
					add_culture_minority_effect = {
						CULTURE = culture:ashkenazi
						SIZE = small
						COUNTY = this
					}
					add_faith_minority_effect = {
						FAITH = faith:rabbinism
						SIZE = small
						COUNTY = this
					}
					if = {
						limit = { NOT = { faith.religion = religion:judaism_religion } }
						
						add_to_variable_list = {
							name = faiths_converted_in_the_county
							target = faith:rabbinism
							years = 40
						}
						save_scope_as = patron_county
						faith:rabbinism = {
							set_variable = {
								name = associated_county
								value = scope:patron_county
							}
							set_variable = {
								name = associated_culture
								value = culture:ashkenazi
							}
							if = {
								limit = {
									NOT = { has_variable = converted_development }
								}
								set_variable = {
									name = converted_development
									value = scope:patron_county.convert_faith_development_value
								}
							}
							else = {
								change_variable = {
									name = converted_development
									add = scope:patron_county.convert_faith_development_value
								}
							}
						}
					}
				}
				
				#Flip tzarfati minorities in counties that speak the same language
				hidden_effect = {
					every_county = {
						limit = {
							county_has_specific_minority_culture_trigger = { CULTURE = culture:tzarfati }
							culture = { has_same_language_group_as_trigger = { TARGET = scope:patron_culture } }
						}
						if = {
							limit = { county_has_specific_large_minority_culture_trigger = { CULTURE = culture:tzarfati } }
							remove_list_variable = {
								name = culture_minorities_large
								target = culture:tzarfati
							}
							add_to_variable_list = {
								name = culture_minorities_large
								target = culture:ashkenazi
							}
						}
						else = {
							remove_list_variable = {
								name = culture_minorities_small
								target = culture:tzarfati
							}
							add_to_variable_list = {
								name = culture_minorities_small
								target = culture:ashkenazi
							}
						}
					}
				}
				
				every_ruler = {
					limit = {
						OR = {
							broadly_jewish_character_trigger = yes
							# Courtiers spawn in this region
							capital_province = {
								OR = {
									geographical_region = world_europe
									geographical_region = world_asia_minor
								}
							}
						}
					}
					trigger_event = bp_culture_emergence_events.1100
				}
			}
		}
	}
}
