on_combat_start = {
	on_actions = { battle_of_tolbiac_dispatch }
}

battle_of_tolbiac_dispatch = {
	trigger = {
		NOT = { has_global_variable = battle_of_tolbiac_happened }
		NOT = { has_global_variable = battle_of_tolbiac_in_progress } #Block potential multiple simultaneous instances
		NOT = { combat_attacker.side_primary_participant = { has_character_flag = battle_of_tolbiac_happened_personal } }
		NOT = { combat_defender.side_primary_participant = { has_character_flag = battle_of_tolbiac_happened_personal } }
	}
	effect = {
		set_global_variable = battle_of_tolbiac_in_progress
		#Pick a side, attacker first
		combat_attacker.side_primary_participant = { save_temporary_scope_as = the_attacker }
		combat_defender.side_primary_participant = { save_temporary_scope_as = the_defender }
		if = {
			limit = {
				battle_of_tolbiac_character_trigger = {
					CHAR_1 = scope:the_attacker
					CHAR_2 = scope:the_defender
				}
				battle_of_tolbiac_side_trigger = {
					SIDE_1 = combat_attacker
					SIDE_2 = combat_defender
				}
			}
			scope:the_attacker = {
				save_scope_as = the_character_in_question
				trigger_event = { on_action = battle_of_tolbiac_event_fire }
			}
		}
		else_if = {
			limit = {
				battle_of_tolbiac_character_trigger = {
					CHAR_1 = scope:the_defender
					CHAR_2 = scope:the_attacker
				}
				battle_of_tolbiac_side_trigger = {
					SIDE_1 = combat_defender
					SIDE_2 = combat_attacker
				}
			}
			scope:the_defender = {
				save_scope_as = the_character_in_question
				trigger_event = { on_action = battle_of_tolbiac_event_fire }
			}
		}
	}
}

battle_of_tolbiac_event_fire = {
	effect = {
		#Get the faith, ordered from Spouse, then Concubines, then counties
		scope:the_character_in_question = {
			ordered_spouse = {
				limit = { this.faith = { NOT = { has_doctrine_parameter = unreformed } } }
				order_by = "reverse_opinion(scope:the_character_in_question)"
				this.faith = { save_scope_as = the_tolbiac_faith }
			}
			if = {
				limit = { NOT = { exists = scope:the_tolbiac_faith } }
				ordered_concubine = {
					limit = { this.faith = { NOT = { has_doctrine_parameter = unreformed } } }
					order_by = "reverse_opinion(scope:the_character_in_question)"
					this.faith = { save_scope_as = the_tolbiac_faith }
				}
			}
			if = {
				limit = { NOT = { exists = scope:the_tolbiac_faith } }
				if = {
					limit = { NOT = { this.capital_county.faith = { has_doctrine_parameter = unreformed } } }
					this.faith = { save_scope_as = the_tolbiac_faith }
				}
				else = {
					random_realm_county = {
						limit = { NOT = { this.faith = { has_doctrine_parameter = unreformed } } }
						this.faith = { save_scope_as = the_tolbiac_faith }
					}
				}
			}
		}
		#Actually fire the event if a scope can be found; final check for validity
		if = {
			limit = { exists = scope:the_tolbiac_faith }
			#Simplify the references
			set_variable = { name = the_tolbiac_faith value = scope:the_tolbiac_faith }
			random = {
				chance = 0.5 #1-in-200 combats where outnumbered
				#Additive modifiers
				modifier = { #Cynicism helps a good bit
					has_trait = cynical
					add = 0.5
				}
				#As does being sinful; the more sinful you are, the more likely your new faith will accept you
				modifier = {
					num_sinful_traits > 0
					add = 0.25
				}
				modifier = {
					num_sinful_traits > 1
					add = 0.25
				}
				modifier = {
					num_sinful_traits > 2
					add = 0.25
				}
				
				#Multiplicative modifiers
				modifier = {
					has_trait = legendary_leader
					factor = 3
				}
				modifier = { #Possessed makes it more likely
					OR = {
						has_trait = possessed_1
						has_trait = possessed_genetic
					}
					factor = 2
				}
				modifier = { #Zealous turns the odds way down, but still *could* happen (St. Paul, Clovis, &c.)
					has_trait = zealous
					factor = 0.1
				}
				
				#Net math: Cynical Legendary Leader (Clovis) will fire it once every 50 battles in which they are loosing
				#Maximal Odds (super sinful, legendary, possessed): 9% of the time (when outnumbered)
				#Minimal Odds: (zealous and nothing else): 1-in-2000 odds (when outnumbered)
				
				trigger_event = battle_of_tolbiac.0001
			}
		}
	}
}

on_combat_end_winner = {
	on_actions = {
		battle_of_tolbiac_global_cleanup
		battle_of_tolbiac_conclusion_winner
	}
}
on_combat_end_loser = {
	on_actions = {
		battle_of_tolbiac_global_cleanup
		battle_of_tolbiac_conclusion_loser
	}
}

battle_of_tolbiac_global_cleanup = {
	trigger = { has_global_variable = battle_of_tolbiac_in_progress }
	effect = { remove_global_variable = battle_of_tolbiac_in_progress }
}

battle_of_tolbiac_conclusion_winner = {
	trigger = { this.side_primary_participant = { has_character_flag = battle_of_tolbiac_in_progress_char } }
	effect = {
		every_side_participant = {
			limit = { has_character_flag = battle_of_tolbiac_in_progress_char }
			remove_character_flag = battle_of_tolbiac_in_progress_char
			trigger_event = { id = battle_of_tolbiac.0010 }
		}
	}
}
battle_of_tolbiac_conclusion_loser = {
	trigger = { this.side_primary_participant = { has_character_flag = battle_of_tolbiac_in_progress_char } }
	effect = {
		every_side_participant = {
			limit = { has_character_flag = battle_of_tolbiac_in_progress_char }
			remove_character_flag = battle_of_tolbiac_in_progress_char
			trigger_event = { id = battle_of_tolbiac.0020 }
		}
	}
}
