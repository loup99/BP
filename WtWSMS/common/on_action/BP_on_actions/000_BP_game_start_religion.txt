#This file should be for _global_ game religious start aspects.
#Year-dependent game start effects should be in their own folder in the by-year section
#So by-year effects will apply _after_ this file

on_game_start = {
	on_actions = { bp_game_start_religion }
}

on_game_start_after_lobby = {
	on_actions = { bp_game_start_religion_after_lobby }
}

bp_game_start_religion = {
	on_actions = {
		#Religious Stuff
		override_vanilla_names
		update_rite_doctrines_and_tenets

		convert_christian_chars_to_rites
		swap_tenets_based_on_dlc
		set_doctrines_by_date
		generate_bishops_for_extant_dioceses
	}
}

bp_game_start_religion_after_lobby = {
	on_actions = {
		move_cathedrals_to_correct_baronies_and_set_buildings
	}
}

BP_set_religion_effects_by_date = {
	effect = {
		#Mazdakism as a strain of mainline Zoroastianism before Sukhra's death, so this is more Zaradushtism
		#See Crone's "Kavad's Heresy and Mazdak's Revolt" for a thorough treatment
		if = {
			limit = { current_year < 493 }
			rite:mazdakism = {
				#Was less ascetic and more "life is to be enjoyed" (although "Hedonistic" might be too far)
				change_rite_doctrine = doctrine_theocracy_temporal
				change_rite_doctrine = doctrine_gender_male_dominated
				change_rite_doctrine = doctrine_major_branch_dorustdin
				change_rite_doctrine = doctrine_clerical_function_taxation
				change_rite_doctrine = doctrine_ritual_vegetarianism_partial
				change_rite_doctrine = doctrine_consanguinity_unrestricted #Mazdakism breaks without it
				
				remove_rite_tenet = tenet:tenet_asceticism
				remove_rite_tenet = tenet:tenet_communal_possessions
				remove_rite_tenet = tenet:tenet_pacifism
				
				add_rite_tenet = tenet:tenet_hedonistic
				add_rite_tenet = tenet:tenet_communal_identity
				add_rite_tenet = tenet:tenet_ritual_celebrations
				
				this.faith = {
					set_tenet_status = {
						tenet = tenet:tenet_communal_possessions
						status = unknown
					}
					set_tenet_status = {
						tenet = tenet:tenet_pacifism
						status = unknown
					}
				}
			}
		}
	}
}
