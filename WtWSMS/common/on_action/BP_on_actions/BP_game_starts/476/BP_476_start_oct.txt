on_game_start = { on_actions = { bp_game_start_476_oct } }

bp_game_start_476_oct = {
	trigger = { current_date = 476.10.15 }
	on_actions = {
		#Characters
		BP_character_events_476_oct
		BP_setup_legitimacy_476_oct
		BP_setup_realm_laws_476_oct_DLC
		BP_setup_vassal_contracts_476_oct_DLC
		BP_remove_claims_476_oct
		BP_senate_476_oct
		#culture
		BP_starting_culture_acceptance_476_oct
	}
}

BP_character_events_476_oct = {
	effect = {
		#Odoacer gets some peace treaties
		character:55 = {
			add_legitimacy = 850
			#Visigoths
			add_truce_both_ways = { character = character:30 years = 10 name = odoacer_peace_visigoths }
			add_truce_both_ways = { character = character:190066 years = 10 name = odoacer_peace_visigoths }
			#Vandals
			add_truce_both_ways = { character = character:93 years = 10 name = odoacer_peace_vandals }
			add_truce_both_ways = { character = character:92 years = 10 name = odoacer_peace_vandals }
			#WRE
			add_truce_both_ways = { character = character:175110 years = 10 name = odoacer_peace_wre }
		}
		
		#Diarchies
		
		#Entrench Diarchies and adjust senate approval
		trigger_event = { id = BP_maintenance_events.1000 }
	}
}
BP_setup_legitimacy_476_oct = {
	effect = {
		#Odoacer should be semi-legitimate
		title:h_italy.holder = { add_legitimacy = -5000 }
		title:h_italy.holder = { add_legitimacy = 500 }
		#Zeno should be fairly illegitimate
		title:h_eastern_roman_empire.holder = { add_legitimacy = -5000 }
		title:h_eastern_roman_empire.holder = { add_legitimacy = 500 }
	}
}

		}
	}
}

	}
}
BP_senate_476_oct = {
	effect = {
		#Western Emperor and Consuls
		character:RufiusAggeriusFestus = { #Rufius Aggerius Festus
			add_trait = senator_west
			add_character_modifier = roman_consul_west
			set_global_variable = {
				name = west_consul
				value = this
			}
		}
		character:55 = { #Odoacer
			set_variable = already_asked_the_senate
			#Approved by the Senate and can't ask again
			#add_character_modifier = approved_by_the_senate
		}
		#Eastern Emperor and Consuls
		character:70515 = { #Zeno
			add_trait = senator_east
			add_character_modifier = roman_consul_east
			set_global_variable = {
				name = east_consul
				value = this
			}
			#Approved by the Senate and can't ask again
			set_variable = already_asked_the_senate
			#add_character_modifier = approved_by_the_senate
		}
	}
}

			}
		}
	}
}

BP_setup_roman_council_476_oct = {
	trigger = { BP_has_dlc_necessary_for_roman_admin_governments_trigger = yes }
	effect = {
		#Set Various Roman Offices
		title:h_western_roman_empire.holder ?= {
			#Populate Nepos' realm with a few characters so he can fill his council
			add_more_wre_courtiers_effect = yes
		}
		title:h_western_roman_empire.holder = { fire_councillor_skip_effects = cp:councillor_chancellor_roman }
		
		title:h_eastern_roman_empire.holder ?= {
			make_councilor_of_type_if_not_already_effect = {
				TARGET_CHAR = title:e_magister_military_ere.holder
				POSITION = councillor_marshal_roman
			}
			make_councilor_of_type_if_not_already_effect = {
				TARGET_CHAR = title:e_magister_offices_ere.holder
				POSITION = councillor_chancellor_roman
			}
			assign_councillor_type = {
				type = councillor_chancellor_roman
				target = character:PLRE2_Felix_17
				remove_existing_councillor = yes
			}
		}
		
		#Italia
		title:h_italy.holder = {
			make_councilor_of_type_if_not_already_effect = {
				TARGET_CHAR = title:e_magister_offices_italy.holder
				POSITION = councillor_chancellor_roman
			}
			make_councilor_of_type_if_not_already_effect = {
				TARGET_CHAR = title:e_magister_military_italy.holder
				POSITION = councillor_marshal_roman
			}
		}
	}
}
