bp_game_start_476 = {
	trigger = { current_date = 476.2.2 }
	on_actions = {
		#Characters
		BP_character_events_476_feb
		BP_character_events_476_feb_DLC
		BP_setup_legitimacy_476_feb
		BP_setup_realm_laws_476_feb_DLC
		BP_starting_state_faiths_476_feb_DLC
		BP_setup_landless_chars_476_feb
		BP_senate_476_feb
	}
}

BP_character_events_476_feb = {
	effect = {
		#Odoacer should be his house head; being the younger sibling and landless blocks that
		character:55 = {
			this.house = { set_house_head = prev }
		}
		
		#Diarchies
		
		#Ensure Orestes is Regent for Romulus and make it rather entrenched
		character:96054 = { designate_diarch = character:96055 }
		remove_generated_diarch_consequences_effect = {
			NEW_DIARCH = character:96055
			LIEGE = character:96054
		}
		#Romulus is not crowned at his age, but give him a smidge of legitimacy
		if = {
			limit = { has_dlc_feature = coronations }
			character:96054 = { add_realm_law_skip_effects = uncrowned }
		}
		#Give Romulus some influence so he can actually call some troops
		character:96054 = { change_influence = romulus_augustulus_at_start_influence_value }
		
		#Entrench Diarchies and adjust senate approval
		trigger_event = { id = BP_maintenance_events.1000 days = 1 }
	}
}
