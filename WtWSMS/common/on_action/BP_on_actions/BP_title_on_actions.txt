# A title is destroyed
# root = the holder before destruction
# scope:landed_title = the title that is being destroyed
# Does *not* provide scope:transfer_type
on_title_destroyed = {
	events = {}
	on_actions = {
		BP_handle_independent_roman_duchies
	}
}

BP_handle_independent_roman_duchies = {
	effect = {
		# If the title of an independent Roman Dynastic ruler is destroyed,
		# let's make sure we handle any stray Roman Dynastic vassals
		if = {
			limit = {
				government_has_flag = government_only_valid_for_kingdoms_empires
				government_has_flag = government_uses_roman_dynastic_authority
				is_independent_ruler = yes
				NOT = {
					any_held_title = {
						tier > tier_duchy
						NOT = { this = scope:landed_title }
					}
				}
			}
			convert_to_subroman_government_effect = yes
			every_vassal_or_below = {
				limit = { government_has_flag = government_uses_roman_dynastic_authority }
				convert_to_subroman_government_effect = yes
			}
		}
	}
}

# root = the new holder
# scope:title = the title that changes hands
# scope:previous_holder = previous holder. Might be dead
on_title_gain = {
	events = {}
	on_actions = {
		BP_handle_centralized_realm_stuff
		BP_apply_title_laws
		#Specific other on-actions
		BP_add_theocratic_traits
		BP_hegemonic_collapse
		BP_check_independent_roman_polities
		BP_fix_tribal_government_grants
		BP_destroy_multiple_nf_titles
	}
}

BP_check_independent_roman_polities = {
	effect = {
		#Fails the check properly if not delayed, so we have to fire via event
		trigger_event = { on_action = BP_handle_independent_roman_polities delayed = yes }
	}
}

BP_handle_independent_roman_polities = {
	trigger = {
		is_independent_ruler = yes
		government_has_flag = government_only_valid_for_empires_hegemonies
		has_main_roman_government_trigger = yes
		
	}
	effect = {
		# If the title of an independent Roman Dynastic/Admin ruler is destroyed,
		# let's make sure we handle any stray Roman Dynastic/Admin vassals
		if = {
			limit = { primary_title.tier < tier_empire }
			root = { convert_to_subroman_government_effect = yes }
			every_vassal_or_below = {
				limit = { has_main_roman_government_trigger = yes }
				convert_to_subroman_government_effect = yes
			}
		}
	}
}

BP_handle_centralized_realm_stuff = {
	trigger = {
		government_has_flag = government_is_centralized
		is_independent_ruler = yes
		highest_held_title_tier >= tier_empire
		scope:title.tier > primary_title.tier
	}
	effect = {
		# Ensure Emperor keeps old realm capital in admin realms & apply unique domicile
		# building bonus if appropriate
		if = { # Ensure we keep the old realm capital
			limit = {
				scope:title = {
					tier >= tier_empire
					NOT = { title_capital_county = root.capital_county }
					# Constantinople must have been inherited
					title_capital_county.holder = root
				}
			}
			set_realm_capital = scope:title.title_capital_county
		}
	}
}

BP_apply_title_laws = {
	effect = {
		if = {
			limit = {
				scope:title.tier = tier_empire
				scope:transfer_type = flag:created
			}
			#Roman Governments
			if = {
				limit = { government_has_flag = empires_use_imperial_elective_succession }
				scope:title = { add_title_law = imperial_elective_succession_law }
			}
			#Iranian Governments
			if = {
				limit = { government_has_flag = empires_use_mahestan_succession }
				scope:title = { add_title_law = mahestan_elective_succession_law }
			}
		}
	}
}

BP_add_theocratic_traits = {
	trigger = {
		government_has_flag = government_is_theocracy
	}
	effect = {
		bp_setup_priest_traits_effect = yes
	}
}

BP_hegemonic_collapse = {
	trigger = {
		#Not China; it has its own rules
		NOT = { scope:title = title:h_china }
		#Only Hegemonies
		scope:title.tier = tier_hegemony
		#Your territory must be too small for hegemonic pretensions
		realm_size < 50
		#Must not be an accepted transfer type
		NOT = { scope:transfer_type = flag:created }
		NOT = { scope:transfer_type = flag:granted }
		NOT = { scope:transfer_type = flag:abdication }
		NOT = { scope:transfer_type = flag:faction_demand }
		NOT = { scope:transfer_type = flag:usurped }
	}
	effect = {
		if = { #Roman Empires are special
			limit = {
				OR = {
					scope:title = title:h_western_roman_empire
					scope:title = title:h_eastern_roman_empire
					scope:title = title:h_roman_empire
				}
			}
			scope:title = { save_scope_as = the_hegemony }
			root = { save_scope_as = the_bag_holder }
			scope:previous_holder ?= { save_scope_as = last_hegemon }
			trigger_event = { id = bp_hegemonic_collapse.0010 }
		}
		else = { #All other Hegemonies
			scope:title = { save_scope_as = the_hegemony }
			root = { save_scope_as = the_bag_holder }
			scope:previous_holder = { save_scope_as = last_hegemon }
			trigger_event = { id = bp_hegemonic_collapse.0001 }
		}
		# Add chronicle entry
		situation:BP_chronicle = {
			record_situation_special_event = {
				key = situation_chronicle_title_destroyed_death_entry
				phase = major_title_destroyed
				variables = {
					char = scope:previous_holder
					title = scope:title
				}
			}
		}
	}
}

BP_fix_tribal_government_grants = {
	trigger = {
		OR = {
			scope:transfer_type = flag:granted
			scope:transfer_type = flag:appointment
		}
	}
	effect = {
		if = {
			limit = {
				NOT = { government_has_flag = government_is_quasi_feudal }
				scope:title = { is_landless_type_title = no }
				scope:title.title_province = { has_holding_type = tribal_holding }
				any_held_title = {
					count <= 1
					this.tier = tier_county
					is_landless_type_title = no
				}
			}
			if = {
				limit = { culture = { no_longer_in_tribal_era_trigger = no } }
				change_government = tribal_government
			}
			else = { change_government = high_tribal_government }
		}
	}
}

BP_destroy_multiple_nf_titles = {
	trigger = { scope:title = { is_noble_family_title = yes } }
	effect = {
		#Don't accumulate NF titles
		if = {
			limit = {
				any_held_title = {
					is_noble_family_title = yes
					NOT = { this = scope:title }
				}
			}
			destroy_title = scope:title
		}
	}
}

# A vassal changes their liege
# root = vassal changing their status
# scope:old_liege = the previous liege of the vassal
# It is also fired for rulers gaining independence, those can be identified by checking current liege
on_vassal_change = {
	on_actions = {
		BP_degrade_roman_government
		BP_cleanup_landless_chars_under_non_admin
		BP_completely_landless_realm_vassal_change
	}
}

BP_completely_landless_realm_vassal_change = {
	trigger = {
		scope:old_liege ?= {
			is_independent_ruler = yes
			government_allows = administrative
			is_landed = no
			trigger_if = {
				limit = { any_vassal = { count > 0 } }
				any_vassal_or_below = { count < 1 is_landed = yes primary_title.tier >= tier_county }
			}
		}
	}
	effect = {
		scope:old_liege ?= {
			if = {
				limit = { any_vassal = { count > 0 } }
				every_vassal_or_below = { every_held_title = { prev = { destroy_title = prev } } }
			}
			every_held_title = { prev = { destroy_title = prev } }
		}
	}
}


BP_degrade_roman_government = {
	effect = {
		### If an admin ruler becomes independent, who are not of a high enough rank, changes government
		if = {
			limit = {
				liege = root # I.e. root is independent
				government_has_flag = government_only_valid_for_kingdoms_empires
				government_has_flag = government_uses_roman_dynastic_authority
				highest_held_title_tier < tier_kingdom
			}
			convert_to_subroman_government_effect = yes
		}
	}
}

BP_cleanup_landless_chars_under_non_admin = {
	trigger = {
		root.top_liege != root
		NOT = { root.top_liege ?= { government_allows = administrative } }
		NOT = { any_held_title = { count > 0 is_landless_type_title = no } }
		NOT = { any_held_title = { count > 0 is_head_of_faith = yes } }
	}
	effect = {
		if = {
			limit = { is_ai = no }
			create_landless_adventurer_title_effect = {
				REASON = flag:deposition
				FLAVOR_CHAR = this
			}
		}
		else = {
			every_held_title = {
				limit = { is_landless_type_title = yes }
				prev = { destroy_title = prev }
			}
		}
	}
}

# A character gains a vassal
# root = character gaining vassal
# scope:vassal = vassal being gained
# scope:old_liege = the previous liege of the vassal. Might be the null character (vassal used to be a non-ruler or independent), so make sure to use "exists" checks where relevant
on_vassal_gained = {
	on_actions = {
		BP_subroman_to_roman
	}
}

BP_subroman_to_roman = {
	trigger = {
		government_has_flag = is_a_roman_government
		NOT = { government_has_flag = is_a_transitory_roman_government }
		scope:vassal = { government_has_flag = is_a_transitory_roman_government }
	}
	effect = {
		#Sub-Romans change government to roman if they get vassalized
		scope:vassal = {
			convert_to_roman_government_effect = yes
			every_vassal_or_below = {
				limit = { government_has_flag = is_a_transitory_roman_government }
				convert_to_roman_government_effect = yes
			}
		}
	}
}

# A title is lost by a character
# root = the old holder
# scope:title = the title that changes hands
# scope:new_holder = new holder.
on_title_lost = {
	on_actions = {
		BP_completely_landless_realm
	}
}

##### Generic On Actions

BP_completely_landless_realm = {
	trigger = {
		is_independent_ruler = yes
		government_allows = administrative
		is_landed = no
		scope:title = { is_head_of_faith = no }
		trigger_if = {
			limit = { any_vassal = { count > 0 } }
			any_vassal_or_below = { count < 1 is_landed = yes primary_title.tier >= tier_county }
		}
	}
	effect = {
		if = {
			limit = { any_vassal = { count > 0 } }
			every_vassal_or_below = { every_held_title = { prev = { destroy_title = prev } } }
		}
		every_held_title = { prev = { destroy_title = prev } }
	}
}

on_realm_capital_change = {
	on_actions = { bp_consume_capital_movement_variables }
}

# Called when realm capital changes. It can be manual move, loss in conquest, move after inheritance,
# move when granted a better title, becoming landed for the first time
# root: New realm capital barony
# scope:old_capital - previous capital barony. Can be empty, can be owned by someone else
bp_consume_capital_movement_variables = {
	effect = {
		if = {
			limit = {
				holder = { has_variable = migration_can_move_capital }
				NOT = { holder = { has_variable = block_immediate_variable_removal } }
			}
			holder = { remove_variable = migration_can_move_capital }
		}
	}
}
