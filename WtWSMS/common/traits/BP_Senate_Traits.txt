	shown_in_ruler_designer = yes
	flag = initial_legitimacy_boost

		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = this } }
				desc = trait_senator_west_desc
			}

			triggered_desc = {
				trigger = { this.dynasty = { has_dynasty_modifier = roman_senate_family_west } }
				desc = trait_senator_west_family_desc
			}
			desc = trait_senator_west_character_desc
		}
	}
	
	monthly_prestige = 0.5
	
	culture_modifier = { #NB: Only applies if the char's culture has the associated tradition
		parameter = roman_legal_traits_are_valued
		monthly_prestige_gain_mult = 0.05

		legitimacy_gain_mult = 0.05
	}
	culture_modifier = { #NB: Only applies if the char's culture has the associated tradition
		parameter = roman_popular_opinion_traits_malus
		county_opinion_add = -5
	}
	culture_modifier = { #NB: Only applies if the char's culture has the associated tradition
		parameter = roman_cultural_trait_opinion_bonus
		same_culture_opinion = 5
	}
	
	ruler_designer_cost = 20
	
	flag = roman_senator
}
