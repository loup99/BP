#Pass Law Values
gold_treasury_cost_from_realm_size = {
	value = 0
	every_realm_county = {
		add = {
			value = this.development_level
			min = 5
		}
	}
	if = {
		limit = { NOT = { culture = { has_innovation = innovation_leges_barbarorum } } }
		multiply = 2.0
		desc = NOT_HAS_INNOVATION_LEGES_BARBARORUM
	}
	if = {
		limit = { NOT = { culture = { has_innovation = innovation_mundium } } }
		multiply = 1.5
		desc = NOT_HAS_INNOVATION_MUNDIUM
	}
	ceiling = yes
}

#Tribal Sub-Roman Decision
number_of_counties_that_are_held_by_you_or_a_subroman = {
	value = 0
	every_realm_county = {
		limit = {
			OR = {
				this.holder = root
				this.holder = { government_has_flag = is_a_roman_government }
			}
		}
		add = 1
	}
}
convert_tribal_to_subroman_government_decision_realm_size = {
	value = 0
	every_realm_county = { limit = { is_landless_type_title = no } add = 1 }
}
number_of_counties_that_are_held_by_you_or_a_subroman_ratio = {
	value = number_of_counties_that_are_held_by_you_or_a_subroman

}
#Culture Creation
number_of_counties_in_region = {
	value = 0
	save_temporary_scope_as = geo_region
	scope:culture_in_question = {
		every_culture_county = {
			limit = { title_capital_county.title_province = { geographical_region = scope:geo_region } }
			add = 1
			#The culture former helps overcome any ties
			if = {
				limit = { this.holder = scope:the_culture_former }
				add = 0.1
			}
		}
	}
}
