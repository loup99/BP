migration_war_effective_fame_level = {
	value = 0
	add = this.prestige_level
	if = {
		limit = { culture = { has_cultural_parameter = cb_lower_prestige_level_requirement } }
		add = 1 #Yes, this is correct; lowering prestige requirement is effectively making your fame level 1 higher
	}
	if = {
		limit = { any_character_trait = { has_trait_flag = one_effective_fame_level_for_migrations } }
		add = 1
	}
	min = low_prestige_level
	max = max_prestige_level
}

migration_chance_of_flip_culture_faith = {
	value = 30
	subtract = {
		value = scope:the_county_development
		subtract = 5
		multiply = 2
	}
	if = {
		limit = {
			OR = {
				scope:the_county.title_province = { has_holding_type = tribal_holding }
				scope:the_county.title_province = { has_holding_type = nomad_holding }
				scope:the_county.title_province = { has_holding_type = herder_holding }
			}
		}
		multiply = 2
	}
	if = {
		limit = { scope:the_county_culture = { num_discovered_innovations_in_era = { target = culture_era_classical_antiquity value > 0 } } }
		subtract = 5
	}
	if = {
		limit = { scope:the_county_culture = { num_discovered_innovations_in_era = { target = culture_era_late_antiquity value > 0 } } }
		subtract = 5
	}
	if = {
		limit = { scope:the_county_culture = { num_discovered_innovations_in_era = { target = culture_era_early_medieval value > 0 } } }
		subtract = 5
	}
	if = {
		limit = { scope:the_county_culture = { num_discovered_innovations_in_era = { target = culture_era_high_medieval value > 0 } } }
		subtract = 5
	}
	if = {
		limit = { scope:the_county_culture = { num_discovered_innovations_in_era = { target = culture_era_late_medieval value > 0 } } }
		subtract = 5
	}
	#And if we hit the limit, no more chances get applied
	if = {
		limit = { scope:the_attacker.current_number_of_flip_counties >= scope:the_attacker.max_number_of_flip_counties }
		multiply = 0
	}
}
max_number_of_flip_counties = {
	value = 0
	if = {
		limit = { has_variable = number_of_migration_source_counties }
		add = var:number_of_migration_source_counties
	}
}
current_number_of_flip_counties = {
	value = 0
	every_realm_county = {
		limit = {
			this.culture = prev.culture
			this.faith = prev.faith
		}
		add = 1
	}
}
