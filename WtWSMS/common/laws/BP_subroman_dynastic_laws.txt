#Somewhat unique:  you can't have better than your liege, but it's not locked to the liege's
subroman_dynastic_aristocracy_law = {
	default = subroman_dynastic_aristocracy_law_0
	cumulative = no
	
	flag = subroman_administrative_aristocracy_law
	
	subroman_dynastic_aristocracy_law_0 = {
		modifier = {
			#monthly_influence_mult = 0.0
			title_creation_cost_mult = 0.0
			monthly_prestige_gain_mult = 0.0
		}
		flag = succession_weight_dynastic_1
		flag = governor_efficency_weight_1
		flag = subroman_administrative_aristocracy_law_0
		
		can_keep = {
			bp_realm_law_use_subroman_dynastic_aristocracy_trigger = yes
		}
		
		can_pass = {
			subroman_TYPE_aristocracy_law_can_pass_trigger = {
				TYPE = dynastic
				LESS_ARISTOCRATIC_LEVEL = 0
				CURRENT_ARISTOCRATIC_LEVEL = 0
				MORE_ARISTOCRATIC_LEVEL = 1
				USES_INFLUENCE = no
			}
		}
		
		on_pass = {
			subroman_TYPE_aristocracy_law_can_pass_effect = {
				TYPE = dynastic
				LESS_ARISTOCRATIC_LEVEL = 0
				CURRENT_ARISTOCRATIC_LEVEL = 0
				MORE_ARISTOCRATIC_LEVEL = 1
				LEAST_ARISTOCRATIC_LEVEL = 0
				MOST_ARISTOCRATIC_LEVEL = 4
				USES_INFLUENCE = no
			}
		}
		
		ai_will_do = {
			#AI will improve to this law if they're running short on prestige or influence
			if = {
				limit = { has_realm_law = subroman_dynastic_aristocracy_law_1 }
				if = {
					limit = { this.prestige < 0 }
					value = 100
				}
				#else_if = {
				#	limit = { this.influence < 50 }
				#	value = 50
				#}
				else = { value = 0 }
			}
		}
	}
	
	subroman_dynastic_aristocracy_law_1 = {
		modifier = {
			#monthly_influence_mult = 0.10
			title_creation_cost_mult = 0.10
			monthly_prestige_gain_mult = 0.10
		}
		flag = revoke_depose_governors_costs_more_influence_1
		flag = succession_weight_dynastic_2
		flag = governor_efficency_weight_2
		flag = subroman_administrative_aristocracy_law_1
		
		can_keep = {
			bp_realm_law_use_subroman_dynastic_aristocracy_trigger = yes
		}
		
		can_pass = {
			subroman_TYPE_aristocracy_law_can_pass_trigger = {
