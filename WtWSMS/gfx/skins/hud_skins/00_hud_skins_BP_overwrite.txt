norse = {
	is_valid = {
		has_dlc_feature = the_northern_lords
	}
	is_shown = {
		OR = {
			has_culture = culture:norse
			AND = {
				religion = religion:germanic_religion
				##### Changed from Vanilla: Applied to all Germanics
				culture = { has_cultural_parameter = heritage_group_germanic }
			}
		}
	}
}

iberian = {
	is_valid = {
		has_dlc_feature = the_fate_of_iberia
	}
	is_shown = {
		##### Changed from Vanilla: Used for Roman government
		OR = {
			culture = { has_cultural_pillar = heritage_iberian }
			AND = {
				government_has_flag = is_a_roman_government
				NAND = { # Use Byzantine GFX for Romans proper
					has_dlc_feature = roads_to_power
					OR = {
						has_main_roman_government_trigger = yes
						culture = { has_building_gfx = byzantine_building_gfx }
					}
				}
			}
		}
	}
}

iranian = {
	is_valid = {
		has_dlc_feature = legacy_of_persia
	}
	is_shown = {
		OR = {
			culture = { has_cultural_pillar = heritage_iranian }
			AND = {
				culture = { has_cultural_pillar = heritage_turkic }
				NOT = {
					government_has_flag = government_is_nomadic
				}
			}
			##### Changed from Vanilla: Used for Iranian government and culture group
			AND = {
				culture = { has_cultural_parameter = heritage_group_greater_iranian }
				NOT = {
					government_has_flag = government_is_nomadic
				}
			}
			government_has_flag = government_is_iranian_dynastic
		}
	}
}

byzantine = {
	is_valid = {
		has_dlc_feature = roads_to_power
	}
	is_shown = {
		OR = {
			culture = { has_cultural_pillar = heritage_byzantine }
			culture = { has_cultural_pillar = heritage_caucasian }
			##### Changed from Vanilla: Used for Roman government
			government_has_flag = is_a_roman_government
		}
	}
}
